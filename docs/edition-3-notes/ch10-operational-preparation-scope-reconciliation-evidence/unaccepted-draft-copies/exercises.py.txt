def run(mode):
 work=HERE/'runtime'/mode;work.mkdir(exist_ok=False);started=time.monotonic();result={'mode':mode,'passed':False,'scope':'preparation-only loopback rehearsal','started_monotonic':started}
 try:result['observations']=globals()[mode](work);result['passed']=True
 except Exception as error:
  # No arbitrary exception payloads or local credential-path exports.
  result['failure']={'class':type(error).__name__,'location':traceback.extract_tb(error.__traceback__)[-1].lineno}
 finally:
  result['duration_seconds']=time.monotonic()-started;write_json(EVIDENCE/(mode+'.result.json'),result);print(json.dumps(result,indent=2))
 if not result['passed']:raise SystemExit(1)
