"""Prepared only. Import performs no network operation; no browser is launched here."""
import base64,json,urllib.parse
from pathlib import Path
from harness import WS,HERE,scrub,write_json,sha

class CDPCollector:
 def __init__(self,debugger_ws_url,artifact_dir,*,fresh_root_authorization):
  if not fresh_root_authorization:raise PermissionError('fresh root browser authorization required')
  u=urllib.parse.urlsplit(debugger_ws_url)
  if u.scheme!='ws' or u.hostname!='127.0.0.1' or u.port!=9228:raise PermissionError('owned CDP endpoint only')
  self.socket=WS(u.port,u.path);self.directory=Path(artifact_dir);self.directory.mkdir(exist_ok=True);self.serial=0;self.phase='startup';self.events=[];self.closed=False
  for name in ['Runtime.enable','Page.enable','Network.enable','Log.enable']:self.command(name)
  self.command('Emulation.setDeviceMetricsOverride',{'width':1440,'height':900,'deviceScaleFactor':1,'mobile':False})
 def observe(self,event):
  method=event.get('method','');p=event.get('params',{});out={'phase':self.phase,'method':method}
  if method=='Network.webSocketFrameReceived':
   try:m=json.loads(p.get('response',{}).get('payloadData',''))
   except (TypeError,ValueError):return
   t=m.get('type')
   if t not in ['part_delta','part_final','tool_finished','replay_end','turn_started','turn_ended','pause_changed','state_changed','error','ephemeral_queued']:return
   out.update(type=t,part_id=str(m.get('part_id','')),paused=m.get('paused'),state=m.get('state'))
   if t in ('part_delta','part_final'):
    part=m.get('part',{});out.update(kind=m.get('kind',part.get('type')),role=part.get('from',m.get('role')),readable_text=str(part.get('text',m.get('text','')))[:4000])
  elif method=='Runtime.exceptionThrown':out['exception']=str(p.get('exceptionDetails',{}).get('text',''))[:1000]
  elif method=='Runtime.consoleAPICalled':
   if p.get('type') not in ('error','warning'):return
   out.update(level=p.get('type'),arguments=[str(a.get('value',a.get('description','')))[:1000] for a in p.get('args',[])])
  elif method=='Log.entryAdded':
   entry=p.get('entry',{})
   if entry.get('level') not in ('error','warning'):return
   out.update(level=entry.get('level'),message=str(entry.get('text',''))[:1000])
  elif method=='Network.loadingFailed':out.update(error=str(p.get('errorText',''))[:1000],cancelled=p.get('canceled',False),resource_type=p.get('type'))
  elif method in ('Network.webSocketCreated','Network.webSocketClosed','Network.webSocketFrameError'):
   if 'url' in p:
    u=urllib.parse.urlsplit(p['url']);out['endpoint']=u.scheme+'://'+u.netloc+u.path
   if 'errorMessage' in p:out['error']=str(p['errorMessage'])[:1000]
  else:return
  self.events.append(scrub(out));assert len(self.events)<=10000,'bounded collector'
 def command(self,method,params=None):
  self.serial+=1;request_id=self.serial;self.socket.send({'id':request_id,'method':method,'params':params or {}})
  while True:
   message=self.socket.receive()
   if message.get('id')==request_id:
    if 'error' in message:raise RuntimeError('CDP command failed: '+method)
    return message.get('result',{})
   self.observe(message)
 def evaluate(self,expression):
  result=self.command('Runtime.evaluate',{'expression':expression,'returnByValue':True,'awaitPromise':True})
  if result.get('exceptionDetails'):raise RuntimeError('DOM expression failed')
  return result.get('result',{}).get('value')
 def snapshot(self,phase,screenshot=False):
  self.phase=phase;dom=self.evaluate((HERE/'dom-probe.js').read_text());payload={'phase':phase,'dom':dom,'events':list(self.events),'browser_observation':True}
  if screenshot:
   image=self.command('Page.captureScreenshot',{'format':'png','captureBeyondViewport':False});path=self.directory/(phase+'.png');path.write_bytes(base64.b64decode(image['data'],validate=True));payload['screenshot_local_only']={'path':str(path),'sha256':sha(path),'bytes':path.stat().st_size,'portable_text_member':False}
  return write_json(self.directory/(phase+'.json'),payload)
 def close(self):
  if not self.closed:self.socket.close();self.closed=True
