#!/usr/bin/env python3
"""Receipt wrapper for the four explicitly authorized local preparation modes."""
import datetime,hashlib,json,shutil,subprocess,sys,time
from pathlib import Path
HERE=Path(__file__).resolve().parent
CFG=json.loads((HERE/'config.json').read_text());ROOT=Path(CFG['root']);EVIDENCE=Path(CFG['evidence']);ENV=CFG['environment']
def pin(path):
 p=Path(path);b=p.read_bytes();return {'path':str(p),'bytes':len(b),'sha256':hashlib.sha256(b).hexdigest()}
def verify():
 if shutil.disk_usage(ROOT).free<1024**3:raise RuntimeError('free-space floor')
 head=subprocess.check_output(['git','--no-pager','rev-parse','HEAD'],cwd=ROOT,env=ENV,text=True).strip();tree=subprocess.check_output(['git','--no-pager','rev-parse','HEAD^{tree}'],cwd=ROOT,env=ENV,text=True).strip();status=subprocess.check_output(['git','--no-pager','status','--porcelain'],cwd=ROOT,env=ENV,text=True);assert head==CFG['source_head'] and tree==CFG['source_tree'] and status==''
 for f in CFG['source_files']:
  assert pin(ROOT/f['path'])['sha256']==f['sha256']
 for f in CFG['artifacts']+CFG['assets']+CFG['public_inputs']+CFG['prior_members']:
  p=pin(f['path']);assert p['sha256']==f['sha256'] and p['bytes']==f['bytes']
 helpers=[pin(HERE/x) for x in ['harness.py','exercises.py','cdp_collector.py','dom-probe.js','run_rehearsal.py','future-plan.json']]
 return {'source_head':head,'source_tree':tree,'source_status':'clean','artifacts':CFG['artifacts'],'assets':CFG['assets'],'helpers':helpers,'prior_members_verified':len(CFG['prior_members'])}
mode=sys.argv[1];assert mode in ['syntax-python','syntax-dom','offline','cancel','sentinel','negative'];receipt=EVIDENCE/(mode+'.receipt.json');assert not receipt.exists();before=verify();start=time.monotonic();began=datetime.datetime.now(datetime.timezone.utc).isoformat()
if mode=='syntax-python':argv=[sys.executable,'-c',"import ast,pathlib; p=pathlib.Path('.ch10-local/operational-preparation'); [(ast.parse((p/n).read_text(),filename=str(p/n))) for n in ['harness.py','exercises.py','cdp_collector.py','run_rehearsal.py']]; print('Task-local helper syntax only; no modes executed.')"]
elif mode=='syntax-dom':argv=['node','--check',str(HERE/'dom-probe.js')]
else:argv=[sys.executable,str(HERE/'harness.py'),mode]
out=EVIDENCE/(mode+'.stdout.txt');err=EVIDENCE/(mode+'.stderr.txt')
with out.open('wb') as stdout,err.open('wb') as stderr:
 p=subprocess.Popen(argv,cwd=ROOT,env=ENV,stdout=stdout,stderr=stderr,start_new_session=True);rc=p.wait()
import os
try:os.killpg(p.pid,0)
except ProcessLookupError:group_gone=True
else:group_gone=False
after=verify();r={'mode':mode,'argv':argv,'cwd':str(ROOT),'environment':ENV,'began_utc':began,'ended_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'duration_seconds':time.monotonic()-start,'exit_code':rc,'owned_runner_pid_and_pgid':p.pid,'owned_runner_group_gone':group_gone,'before':before,'after':after,'outputs':[pin(out),pin(err)],'free_bytes_after':shutil.disk_usage(ROOT).free,'completed_product_gate':False,'browser_or_paid_observation':False}
receipt.write_text(json.dumps(r,indent=2)+'\n');print(json.dumps({'mode':mode,'exit_code':rc,'duration_seconds':r['duration_seconds'],'owned_runner_group_gone':group_gone,'receipt':str(receipt)},indent=2));assert before==after and group_gone;raise SystemExit(rc)
