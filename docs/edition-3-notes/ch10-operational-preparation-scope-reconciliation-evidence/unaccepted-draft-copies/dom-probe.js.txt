/* Passive Runtime.evaluate expression. No app globals/DOM are modified. */
(() => {
  const visible = node => !!node && !node.hidden && getComputedStyle(node).display !== 'none' && getComputedStyle(node).visibility !== 'hidden' && node.getBoundingClientRect().width > 0 && node.getBoundingClientRect().height > 0;
  const rect = node => { const r=node.getBoundingClientRect(); return {x:r.x,y:r.y,width:r.width,height:r.height}; };
  const cards = [...document.querySelectorAll('article[data-part-id]')].map(node => ({
    part_id:node.dataset.partId, role:node.dataset.role, classes:[...node.classList], title:node.querySelector('h2')?.textContent || '',
    visible:visible(node), rect:rect(node), visible_text:visible(node) ? node.innerText.slice(0,4000) : '',
    readable_dom_text:node.textContent.slice(0,4000), streaming:node.classList.contains('streaming'),
    thinking_expanders:[...node.querySelectorAll('details')].map(x=>({open:x.open,summary:x.querySelector('summary')?.textContent || ''}))
  }));
  const text=document.body.innerText, domText=document.body.textContent;
  const pane = id => { const node=document.getElementById(id); return {id,rect:rect(node),scrollHeight:node.scrollHeight,clientHeight:node.clientHeight,scrollWidth:node.scrollWidth,clientWidth:node.clientWidth}; };
  return {
    url:location.origin+location.pathname,title:document.title,ready_state:document.readyState,
    theme:document.documentElement.dataset.theme,viewport:{width:innerWidth,height:innerHeight,device_pixel_ratio:devicePixelRatio},
    connection:document.getElementById('connection')?.textContent,state:document.getElementById('state')?.textContent,
    paused_visible:visible(document.getElementById('paused')),notice:document.getElementById('notice')?.innerText || '',
    cards,unique_part_ids:new Set(cards.map(c=>c.part_id)).size,
    opaque_marker_visible:['PREP_OPAQUE_SIGNATURE_NOT_VISIBLE','PREP_REDACTED_NOT_VISIBLE'].some(x=>text.includes(x)),
    opaque_marker_in_dom:['PREP_OPAQUE_SIGNATURE_NOT_VISIBLE','PREP_REDACTED_NOT_VISIBLE'].some(x=>domText.includes(x)),
    panes:['chat-scroll','actions-scroll'].map(pane),page_overflow:document.documentElement.scrollWidth>innerWidth
  };
})()
