def scrub(value,secrets=()):

class Ledger:
 """Consumes a phase permit BEFORE connecting; failure never refunds it."""
 def __init__(self,path,phases=('first','resume')):

 def flush(self):write_json(self.path,{'phases':self.used,'active':self.active,'entries':self.entries,'retry_or_refund':False})
 def activate(self,phase):
  with self.lock:
   if phase not in self.used or self.active or self.used[phase]:raise Denied('phase not available')
   if phase=='resume' and self.used.get('first')!=1:raise Denied('first phase required')
   self.phase=phase;self.flush()
 def take(self):
  with self.lock:
   if self.phase is None or self.used[self.phase]>=1:raise Denied('request cap')
   self.used[self.phase]+=1;self.active+=1;entry={'phase':self.phase,'started_monotonic':time.monotonic(),'outcome':'active'};self.entries.append(entry);self.flush();return entry
 def finish(self,entry,outcome):
  with self.lock:
   entry['outcome']=outcome;entry['ended_monotonic']=time.monotonic();self.active-=1;self.flush()
