// Audit-only caller of the unchanged exported historical grader. No protocol shim.
package main

import (
 "fmt"
 "os"
 "path/filepath"
 "github.com/waywardgeek/ensemble/internal/gradeedition2"
)

func main() {
 if len(os.Args) != 5 { panic("driver CORE CLI GUI STATIC") }
 core, cli, gui, static := os.Args[1], os.Args[2], os.Args[3], os.Args[4]
 example, err := gradeedition2.ReadConsumerCase(""); if err != nil { panic(err) }
 media, err := gradeedition2.ReadMediaModels(""); if err != nil { panic(err) }
 report, err := gradeedition2.Chapter9(cli, gui, core, filepath.Join(core,"ch05"), filepath.Join(core,"ch06"), filepath.Join(core,"ch07"), filepath.Join(core,"gui"), static, "claude-sonnet-4-6", "claude-sonnet-4-6", example, media)
 if err != nil { fmt.Fprintln(os.Stderr, err); os.Exit(2) }
 if err := report.WriteJSON(os.Stdout); err != nil { panic(err) }
 if !report.Passed { os.Exit(1) }
}
