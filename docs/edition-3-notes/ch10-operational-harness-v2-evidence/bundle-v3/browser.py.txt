class CDP:
    def __init__(self,run,port,path):
        self.run=run;self.wire=Wire(run,port,path,origin=False);self.next=0
        self.app=[];self.errors=[];self.network=[];self.phase='setup';self.replay_end=0;self.turn_ends=[]
    def observe(self,message):
        method=message.get('method');p=message.get('params',{})
        if method=='Network.webSocketFrameReceived' and p.get('response',{}).get('opcode')==1:
            try:event=json.loads(p['response']['payloadData'])
            except (ValueError,KeyError):raise Refused('non-json-gui-wire') from None
            if event.get('type') in ('part_final','part_delta','tool_dispatched','tool_finished','state_changed','pause_changed','replay_begin','replay_end','current_settings','turn_ended','error'):
                self.app.append(event)
                if event['type']=='replay_end':self.replay_end+=1
                if event['type']=='turn_ended':self.turn_ends.append(event)
                if event['type']=='current_settings':require(event['settings']['tts_enabled'] is False,'read-aloud-enabled')
                if event['type']=='error':self.errors.append({'kind':'gui-wire-error','phase':self.phase})
        elif method=='Runtime.exceptionThrown':
            e=p.get('exceptionDetails',{});self.errors.append({'kind':'runtime-exception','phase':self.phase,'text':e.get('text',''),'description':e.get('exception',{}).get('description','')})
        elif method=='Log.entryAdded' and p.get('entry',{}).get('level')=='error':
            self.errors.append({'kind':'console-error','phase':self.phase,'text':p['entry'].get('text','')})
        elif method=='Network.loadingFailed':
            entry={'kind':'load-failure','phase':self.phase,'error':p.get('errorText'),'canceled':p.get('canceled',False)}
            if self.phase=='restart' and entry['canceled']:entry['expected_owned_navigation_cancellation']=True
            else:self.errors.append(entry)
            self.network.append(entry)
        elif method=='Network.requestWillBeSent':
            url=p.get('request',{}).get('url','');u=urlsplit(url)
            require(u.scheme in ('about','data','blob') or (u.scheme in ('http','ws') and u.hostname=='127.0.0.1' and u.port!=8088),'browser-page-external-request')
            self.network.append({'phase':self.phase,'method':p.get('request',{}).get('method'),'url':url})
    def receive(self):
        message=self.wire.receive();self.observe(message);return message
    def command(self,method,params=None):
        self.next+=1;number=self.next;self.wire.send({'id':number,'method':method,'params':params or {}})
        while True:
            m=self.receive()
            if m.get('id')==number:
                require('error' not in m,'cdp-command');return m.get('result',{})
    def evaluate(self,expression):
        result=self.command('Runtime.evaluate',{'expression':expression,'returnByValue':True,'awaitPromise':True})
        require('exceptionDetails' not in result,'probe-exception')
        return result.get('result',{}).get('value')
    def wait(self,predicate):
        while not predicate():self.receive()
    def close(self):self.wire.close()

class Browser(GUI):
    def __init__(self,run,proxy):
        super().__init__(run,proxy);require(run.mode=='browser','browser-mode')
        self.chrome=None;self.cdp=None;self.profile=run.root/'profile';self.target=None;self.last_dom=None
    def ownership(self):
        self.run.check(entering=True);require(self.chrome[0].poll() is None,'chrome-exited')
        pids,rows=self.run.process_tree(self.chrome[0].pid)
        require('--user-data-dir='+str(self.profile) in rows and '--remote-debugging-port=0' in rows,'chrome-profile-command')
        bundle=str(Path(self.run.manifest['chrome']['path']).parents[2])
        require(all(bundle in line for line in rows.splitlines() if line.strip()),'chrome-helper-ancestry')
        self.run.listener(pids,self.debug_port)
        self.run.record('chrome-owned-identity',{'executable':self.run.manifest['chrome'],'profile':str(self.profile),'pids':sorted(pids),'process_rows':rows,'debug_port':self.debug_port,'ancestry_observed_only':True})
    def page_list(self):
        self.ownership();self.run.check(entering=True)
        c=http.client.HTTPConnection('127.0.0.1',self.debug_port,timeout=5)
        try:
            c.request('GET','/json/list');r=c.getresponse();require(r.status==200,'cdp-list');return json.loads(r.read())
        finally:c.close()
    def start(self,index):
        self.start_process(index)
        if index==1:
            require(not self.profile.exists(),'profile-reuse');self.profile.mkdir(mode=0o700)
            argv=[self.run.manifest['chrome']['path'],f'--user-data-dir={self.profile}','--remote-debugging-address=127.0.0.1','--remote-debugging-port=0','--no-first-run','--no-default-browser-check','--disable-background-networking','--disable-component-update','--disable-sync','--disable-extensions','--disable-default-apps','--disable-domain-reliability','--mute-audio','--window-size=1440,1000','about:blank']
            self.chrome=self.run.start('owned-chrome',argv,self.run.env)
            active=self.profile/'DevToolsActivePort'
            while not active.is_file():self.run.check();require(self.chrome[0].poll() is None,'chrome-startup-exit');time.sleep(.05)
            require(not active.is_symlink(),'debug-port-symlink');self.debug_port=int(active.read_text().splitlines()[0]);require(self.debug_port!=8088,'forbidden-debug-port')
            pages=[p for p in self.page_list() if p.get('type')=='page'];require(len(pages)==1 and pages[0].get('url')=='about:blank','one-fresh-page')
            self.target=pages[0]['id'];u=urlsplit(pages[0]['webSocketDebuggerUrl']);require(u.hostname=='127.0.0.1' and u.port==self.debug_port and u.scheme=='ws','owned-page-websocket')
            self.cdp=CDP(self.run,self.debug_port,u.path)
            for domain in ('Page','Runtime','Log','Network'):self.cdp.command(domain+'.enable')
        else:
            pages=[p for p in self.page_list() if p.get('type')=='page'];require(len(pages)==1 and pages[0]['id']==self.target,'same-single-page')
        self.cdp.phase='gui-'+str(index);self.cdp.app=[];before=self.cdp.replay_end
        self.cdp.command('Page.navigate',{'url':self.url});self.cdp.wait(lambda:self.cdp.replay_end>before)
        self.events=self.cdp.app;require(not self.cdp.errors,'browser-console-or-load-error')
        self.run.record('browser-page',{'target_id':self.target,'page_count':1,'url':self.url,'index':index,'requests_before_user':self.proxy.used})
    def prompt(self,text):
        before=len(self.cdp.turn_ends)
        js="(() => { const draft=document.getElementById('draft'), form=document.getElementById('composer'); if(!draft||!form) throw new Error('composer missing'); draft.value="+json.dumps(text)+"; draft.dispatchEvent(new Event('input',{bubbles:true})); form.requestSubmit(); return true; })()"
        require(self.cdp.evaluate(js) is True,'composer-submission')
        self.cdp.wait(lambda:len(self.cdp.turn_ends)>before)
        require(not self.cdp.turn_ends[-1].get('error'),'browser-turn-error');self.events=self.cdp.app
    def snapshot(self,phase):
        self.ownership();wire_cards=cards(self.events)
        self.cdp.evaluate('document.fonts.ready.then(() => new Promise(resolve => requestAnimationFrame(() => requestAnimationFrame(() => resolve(true)))))')
        dom=self.cdp.evaluate((HERE/'dom-probe.js').read_text())
        require(dom['url']==self.url and dom['composer_present'] and dom['read_aloud_enabled'] is False and dom['speech_event_count']==0,'browser-page-or-audio-state')
        actual={c['part_id']:c for c in dom['cards']};require(len(actual)==len(dom['cards']) and set(actual)==set(wire_cards),'dom-wire-card-identities')
        require(all(not c['streaming'] for c in actual.values()),'unfinished-dom-card')
        require(not self.cdp.errors,'browser-console-or-load-error')
        if phase=='after-restore-before-user':
            require(self.last_dom is not None,'missing-before-dom')
            before={c['part_id']:(c['role'],c['title'],c['text']) for c in self.last_dom['cards']}
            after={c['part_id']:(c['role'],c['title'],c['text']) for c in dom['cards']}
            require(before==after,'visible-restore-changed-cards')
        if phase=='after-new-final':
            old={c['part_id']:c for c in self.last_dom['cards']}
            require(all(k in actual and actual[k]['text']==v['text'] for k,v in old.items()),'dom-old-cards-overwritten')
            require(any(k not in old and c['role']!='human' and c['text'].strip() for k,c in actual.items()),'dom-new-assistant-card')
        self.cdp.evaluate('document.fonts.ready.then(() => true)')
        image=self.cdp.command('Page.captureScreenshot',{'format':'png','captureBeyondViewport':False})
        require(not self.cdp.errors,'browser-console-or-load-error')
        image_path=self.run.raw/(phase+'.png');image_path.write_bytes(base64.b64decode(image['data'],validate=True))
        evidence={'phase':phase,'dom':dom,'wire_cards':wire_cards,'console_errors':self.cdp.errors,'load_network_observations':self.cdp.network,'screenshot_local_only':pin(image_path),'human_visual_inspection':False}
        write_json(self.run.root/(phase+'-dom.json'),evidence)
        self.run.record('browser-dom-observation',{'phase':phase,'card_ids':sorted(actual),'screenshot_local_only':pin(image_path),'human_visual_inspection':False})
        if phase in ('before-eof','after-restore-before-user'):self.last_dom=dom
        self.run.browser_dom_observed=True
        return wire_cards
    def stop(self):
        self.cdp.phase='restart'
        self.cdp.command('Page.navigate',{'url':'about:blank'})
        self.run.end(self.pair,eof=True)
    def shutdown(self):
        if self.cdp:
            try:
                self.cdp.wire.send({'id':self.cdp.next+1,'method':'Browser.close','params':{}})
                self.run.record('chrome-close-requested',{'target_id':self.target})
            except OSError:pass
            finally:self.cdp.close()
        if self.chrome:
            try:require(self.chrome[0].wait(timeout=3)==0,'chrome-exit')
            except __import__('subprocess').TimeoutExpired:self.run.fail('chrome-graceful-close-timeout')
