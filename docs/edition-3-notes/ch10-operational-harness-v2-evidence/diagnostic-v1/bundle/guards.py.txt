"""Compact, explicitly authorized guard mode. Never run by import or another mode."""
import http.client
import json
from pathlib import Path
import time
from core import HERE, AUDIT, Refused, require, digest, load, write_json, sequence
from provider import Fixture, Proxy

def http_request(proxy,body):
    proxy.run.check(entering=True)
    c=http.client.HTTPConnection('127.0.0.1',proxy.port,timeout=5)
    try:
        c.request('POST','/v1/messages',body=json.dumps(body).encode(),headers={'Content-Type':'application/json'})
        r=c.getresponse();raw=r.read();return r.status,json.loads(raw)
    finally:c.close()

def exercise(run):
    require(run.mode=='guards','guards-only');run.activate();results=[]
    # Actual entry-point subprocesses, not applications. Neither has matching execution authority.
    bad_approval=run.root/'deliberately-invalid-approval.json'
    write_json(bad_approval,{'authority':'not-root-approval'});bad_approval.chmod(0o600)
    for case,seal,reason in [('wrong-pin','0'*64,'seal-pin'),('preauthorization',run.seal,'approval-fields')]:
        root=HERE/'runs'/('cli-rehearsal-'+case+'-'+digest(str(run.root).encode())[:12])
        require(not root.exists(),'guard-probe-root-exists')
        argv=[run.manifest['python']['path'],str(HERE/'launch.py'),'--mode','cli-rehearsal','--run-root',str(root),'--approval',str(bad_approval),'--expect-seal',seal]
        pair=run.start('probe-'+case,argv,run.env);p,rec=pair;p.stdin.close()
        while p.poll() is None:run.check();time.sleep(.02)
        rec['exit_code']=p.wait();output=load(Path(rec['stdout_local']))
        require(rec['exit_code']==65 and output.get('phase')=='preflight' and output.get('reason')==reason and output.get('entry')=='DENIED' and not root.exists(),'entry-denial-not-observed')
        require(output['audit']['credential_open_attempts']==0,'guard-key-open')
        results.append({'case':case,'exit_code':rec['exit_code'],'observed':output,'run_root_not_created':True})
    valid={'model':run.policy['model'],'max_tokens':1024,'stream':False,'messages':[{'role':'user','content':'Synthetic proxy boundary check, not an application conversation.'}]}
    fake=Fixture(run,synthetic=True);bad_model=Proxy(run,1024,fake,guard_case=True);later=[]
    def first_failure():
        status,_=http_request(bad_model,dict(valid,model='deliberately-wrong-model'))
        require(status==502,'wrong-model-status')
        raise Refused('observed-first-phase-failure')
    def forbidden_later_phase():
        later.append('entered');raise Refused('later-phase-was-entered')
    try:sequence([first_failure,forbidden_later_phase])
    except Refused as error:require(str(error)=='observed-first-phase-failure','sequence-error')
    else:raise Refused('sequence-did-not-stop')
    require(not later and bad_model.used==0 and bad_model.connections==0 and not fake.requests,'wrong-model-or-stop-guard')
    results.append({'case':'wrong-model-and-first-failure-stops-sequence','later_phase_entries':0,'facts':bad_model.facts()})
    require(bad_model.close() and fake.close(),'guard-cleanup')
    fake=Fixture(run,synthetic=True);bad_tokens=Proxy(run,4096,fake,guard_case=True)
    status,_=http_request(bad_tokens,dict(valid,max_tokens=4095))
    require(status==502 and bad_tokens.used==0 and bad_tokens.connections==0 and not fake.requests,'wrong-token-guard')
    results.append({'case':'wrong-native-token-limit','facts':bad_tokens.facts()});require(bad_tokens.close() and fake.close(),'guard-cleanup')
    fake=Fixture(run,synthetic=True);capped=Proxy(run,1024,fake,guard_case=True)
    statuses=[http_request(capped,valid)[0] for _ in range(5)]
    require(statuses==[200,200,200,200,429] and capped.used==4 and capped.connections==4 and len(fake.requests)==4,'four-then-fifth-guard')
    results.append({'case':'four-actual-forwards-fifth-denied','statuses':statuses,'facts':capped.facts()});require(capped.close() and fake.close(),'guard-cleanup')
    require(AUDIT.credential_open_attempts==0,'guard-key-open')
    run.observations['guards']=results
