#!/usr/bin/env python3
"""Explicit mode + exact fresh root approval required. No implicit campaign execution."""
import argparse
import json
import sys
from core import AUDIT, MODES, Refused, Run

def main():
    parser=argparse.ArgumentParser()
    parser.add_argument('--mode',choices=MODES,required=True)
    parser.add_argument('--run-root',required=True)
    parser.add_argument('--approval',required=True)
    parser.add_argument('--expect-seal',required=True)
    args=parser.parse_args()
    try:
        run=Run(args.mode,args.approval,args.run_root,args.expect_seal)
    except Exception as error:
        reason=str(error) if isinstance(error,Refused) else 'preflight-invalid-input'
        print(json.dumps({'phase':'preflight','entry':'DENIED','reason':reason,'audit':{'credential_open_attempts':AUDIT.credential_open_attempts,'preentry_denials':AUDIT.preentry_denials}}))
        return 65
    success=False
    try:
        if args.mode=='guards':
            from guards import exercise
            exercise(run)
        else:
            from flows import two_flow
            two_flow(run,'cli' if args.mode in ('cli-rehearsal','paid-cli') else 'gui')
        run.check();success=True
    except Exception as error:
        run.fail(str(error) if isinstance(error,Refused) else 'harness-or-application-exception')
        run.record('exception-class',{'class':type(error).__name__})
    finally:
        result=run.finish(success)
    print(json.dumps({'mode':args.mode,'status':result['status'],'failure':result['failure'],'cleanup_complete':result['cleanup_complete'],'run_root':str(run.root),'duration_seconds':result['duration_seconds'],'credential_opens':result['credential_audit_python_only']['authorized_opens']}))
    return 0 if result['status']=='PASS' else 1

if __name__=='__main__':
    raise SystemExit(main())
