"""Owned loopback WebSocket only; nothing opens at import."""
import base64
import hashlib
import json
import os
import socket
import struct
from core import require

class Wire:
    def __init__(self,run,port,path='/ws',origin=True):
        run.check(entering=True);require(port!=8088 and path.startswith('/') and '\r' not in path and '\n' not in path,'websocket-address')
        self.run=run;self.events=[];self.closed=False
        self.socket=socket.create_connection(('127.0.0.1',port),timeout=5);self.socket.settimeout(.2)
        key=base64.b64encode(os.urandom(16)).decode()
        request=f'GET {path} HTTP/1.1\r\nHost: 127.0.0.1:{port}\r\nUpgrade: websocket\r\nConnection: Upgrade\r\nSec-WebSocket-Key: {key}\r\nSec-WebSocket-Version: 13\r\n'
        if origin:request+=f'Origin: http://127.0.0.1:{port}\r\n'
        self.socket.sendall((request+'\r\n').encode());header=b''
        while not header.endswith(b'\r\n\r\n'):
            require(len(header)<16384,'websocket-header-size');header+=self.read(1)
        expected=base64.b64encode(hashlib.sha1((key+'258EAFA5-E914-47DA-95CA-C5AB0DC85B11').encode()).digest())
        require(b' 101 ' in header.split(b'\r\n')[0] and expected in header,'websocket-upgrade')
    def read(self,n):
        data=b''
        while len(data)<n:
            self.run.check()
            try:chunk=self.socket.recv(n-len(data))
            except socket.timeout:continue
            require(bool(chunk),'websocket-closed');data+=chunk
        return data
    def frame(self,opcode,payload):
        mask=os.urandom(4);n=len(payload)
        length=bytes([0x80|n]) if n<126 else b'\xfe'+struct.pack('!H',n) if n<65536 else b'\xff'+struct.pack('!Q',n)
        self.socket.sendall(bytes([0x80|opcode])+length+mask+bytes(v^mask[i%4] for i,v in enumerate(payload)))
    def send(self,value):self.frame(1,json.dumps(value).encode())
    def receive(self):
        chunks=[];started=False
        while True:
            a,b=self.read(2);op=a&15;require(not b&128,'masked-server-frame');n=b&127
            if n==126:n=struct.unpack('!H',self.read(2))[0]
            elif n==127:n=struct.unpack('!Q',self.read(8))[0]
            require(n<=8*1024*1024,'websocket-frame-size');body=self.read(n)
            if op==9:self.frame(10,body);continue
            if op==10:continue
            require(op!=8,'websocket-closed');require(op in (0,1) and (op!=0 or started),'websocket-opcode');started=True;chunks.append(body)
            require(sum(map(len,chunks))<=8*1024*1024,'websocket-message-size')
            if a&128:
                value=json.loads(b''.join(chunks));self.events.append(value);return value
    def until(self,predicate):
        while True:
            value=self.receive()
            if predicate(value):return value
    def close(self):
        if not self.closed:
            self.closed=True
            try:self.frame(8,struct.pack('!H',1000))
            except OSError:pass
            self.socket.close()
