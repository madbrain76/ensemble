(() => {
  const q = id => document.getElementById(id);
  const rows = [...document.querySelectorAll('article[data-part-id]')].map(node => {
    const r = node.getBoundingClientRect();
    return {part_id: node.dataset.partId, role: node.dataset.role,
      classes: [...node.classList], title: node.querySelector('h2')?.innerText || '',
      text: node.innerText, streaming: node.classList.contains('streaming'),
      details: [...node.querySelectorAll('details')].map(x => ({open:x.open, summary:x.querySelector('summary')?.innerText || ''})),
      viewport_intersection: r.width>0 && r.height>0 && r.right>0 && r.bottom>0 && r.left<innerWidth && r.top<innerHeight,
      bounds: {x:r.x,y:r.y,width:r.width,height:r.height}};
  });
  return {url:location.href,title:document.title,ready:document.readyState,
    connection:q('connection')?.innerText || '',state:q('state')?.innerText || '',
    paused_visible:q('paused') ? !q('paused').hidden : null,
    read_aloud_enabled:q('read-aloud')?.checked,
    microphone_pressed:q('microphone')?.getAttribute('aria-pressed'),
    speech_event_count:Array.isArray(window.speechEvents)?window.speechEvents.length:null,
    composer_present:!!q('composer') && !!q('draft'),
    viewport:{width:innerWidth,height:innerHeight,device_pixel_ratio:devicePixelRatio,
      document_scroll_width:document.documentElement.scrollWidth},cards:rows};
})()
