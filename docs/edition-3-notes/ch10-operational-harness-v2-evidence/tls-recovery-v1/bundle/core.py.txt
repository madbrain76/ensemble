HERE = Path(__file__).resolve().parent
MODES = ('paid-cli',)
KEY_PATH = Path('/Users/bill') / '.cr' / ('settings' + '.json')

class Refused(Exception):
    pass

def require(condition, code):
    if not condition:
        raise Refused(code)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def pin(path):
    p = Path(path)
    require(p.is_file() and not p.is_symlink(), 'nonregular-pin')
    data = p.read_bytes()
    return {'path': str(p), 'bytes': len(data), 'sha256': digest(data)}

def load(path):
    return json.loads(Path(path).read_text(encoding='utf-8'))

def safe_path(path):
    return str(path).replace(str(KEY_PATH), '[APPROVED_CREDENTIAL_SOURCE]').replace('settings' + '.json', '[LOCAL_CONFIG]')

def clean(value, secret=None):
    if isinstance(value, dict):
        return {safe_path(k): clean(v, secret) for k, v in value.items()}
    if isinstance(value, (list, tuple)):
        return [clean(x, secret) for x in value]
    if isinstance(value, str):
        value = safe_path(value)
        if secret:
            value = value.replace(secret, '[REDACTED]')
        return value
    return value

def write_json(path, value, secret=None):
    p = Path(path)
    data = (json.dumps(clean(value, secret), indent=2) + '\n').encode()
    require(b'\0' not in data, 'nul-export')
    if secret:
        require(secret.encode() not in data, 'secret-export')
    p.write_bytes(data)

def checked_json(raw):
    def pairs(items):
        out = {}
        for k, v in items:
            require(k not in out, 'duplicate-json-field')
            out[k] = v
        return out
    return json.loads(raw, object_pairs_hook=pairs)

class Audit:
    """Python-process audit only; not an OS-wide or child-process syscall audit."""
    authorized = False
    key_window = False
    credential_open_attempts = 0
    authorized_credential_opens = 0
    preentry_denials = 0
    def hook(self, event, args):
        if event in ('subprocess.Popen', 'os.system', 'os.posix_spawn', 'socket.__new__') and not self.authorized:
            self.preentry_denials += 1
            raise Refused('preauthorization-entry')
        if event == 'open' and isinstance(args[0], (str, bytes, os.PathLike)):
            name = os.fsdecode(args[0])
            if os.path.abspath(name) == str(KEY_PATH):
                self.credential_open_attempts += 1
                require(self.authorized and self.key_window and self.authorized_credential_opens == 0, 'credential-open-denied')
                self.authorized_credential_opens += 1

AUDIT = Audit()
sys.addaudithook(AUDIT.hook)

def verify_bundle(expected_seal):
    require(re.fullmatch(r'[0-9a-f]{64}', expected_seal or '') is not None, 'seal-pin')
    seal_path = HERE / 'execution-seal.json'
    require(pin(seal_path)['sha256'] == expected_seal, 'seal-pin')
    seal = load(seal_path)
    require(seal['status'] == 'NOTREHEARSED_NOTEXECUTED' and seal['schema'] == 1, 'seal-schema')
    expected_names = {'core.py', 'wire.py', 'provider.py', 'flows.py', 'browser.py', 'guards.py', 'launch.py', 'dom-probe.js', 'policy.json', 'manifest.json', 'fixture.txt', 'approval-schema.json'}
    require({x['name'] for x in seal['files']} == expected_names, 'seal-members')
    for entry in seal['files']:
        got = pin(HERE / entry['name'])
        require(got['bytes'] == entry['bytes'] and got['sha256'] == entry['sha256'], 'bundle-file-pin')
    manifest = load(HERE / 'manifest.json')
    require(len(manifest['source_files']) == 137, 'source-member-count')
    files = manifest['source_files'] + manifest['identity_files'] + manifest['binaries'] + manifest['assets'] + manifest['metadata_executables'] + [manifest['source_manifest'], manifest['chrome'], manifest['python'], manifest['public_ca']]
    for expected in files:
        require(pin(expected['path']) == expected, 'source-or-artifact-pin')
    return manifest, load(HERE / 'policy.json'), pin(HERE / 'policy.json')['sha256']

def preflight(mode, approval_path, run_root, expected_seal):
    require(mode in MODES, 'mode')
    manifest, policy, policy_sha = verify_bundle(expected_seal)
    require(policy['total_request_limit']==3, 'recovery-budget-policy')
    root = Path(run_root)
    require(root.is_absolute() and root.parent == HERE / 'runs' and root.parent.resolve() == root.parent, 'run-root')
    require(re.fullmatch(re.escape(mode) + r'-[A-Za-z0-9_-]{1,40}', root.name) is not None and not root.exists() and not root.is_symlink(), 'one-shot-run-root')
    a_path = Path(approval_path)
    require(a_path.is_file() and not a_path.is_symlink() and a_path.stat().st_uid == os.getuid() and a_path.stat().st_mode & 0o077 == 0, 'approval-file')
    require(a_path.stat().st_size <= 16384, 'approval-size')
    approval = checked_json(a_path.read_bytes())
    fields = {'authority', 'mode', 'policy_sha256', 'seal_sha256', 'run_root', 'issued_utc', 'expires_utc', 'browser_receipt', 'browser_receipt_sha256', 'browser_accepted'}
    require(set(approval) == fields, 'approval-fields')
    require(approval['authority'] == 'root-explicit-ch10-v2-execution' and approval['mode'] == mode and approval['policy_sha256'] == policy_sha and approval['seal_sha256'] == expected_seal and approval['run_root'] == str(root), 'approval-binding')
    try:
        issued = dt.datetime.fromisoformat(approval['issued_utc'])
        expires = dt.datetime.fromisoformat(approval['expires_utc'])
        now = dt.datetime.now(dt.timezone.utc)
        require(issued.utcoffset() == dt.timedelta(0) and expires.utcoffset() == dt.timedelta(0), 'approval-timezone')
        require(issued <= now < expires and 0 < (expires-issued).total_seconds() <= policy['approval_max_seconds'], 'approval-expiry')
    except (TypeError, ValueError):
        raise Refused('approval-time') from None
    if mode == 'paid-cli':
        require(approval['browser_accepted'] is True and isinstance(approval['browser_receipt'], str), 'browser-acceptance-required')
        carry=policy['carried_browser']
        for expected in [carry['browser_receipt'],carry['original_seal'],carry['original_policy'],carry['original_manifest'],carry['prior_paid_failure']]:
            require(pin(expected['path'])==expected,'carried-pin')
        old_seal=load(carry['original_seal']['path']);old_files={x['name']:x for x in old_seal['files']}
        old_manifest=load(carry['original_manifest']['path'])
        require(old_files['manifest.json']['sha256']==carry['original_manifest']['sha256'] and old_files['policy.json']['sha256']==carry['original_policy']['sha256'],'carried-manifest-policy-chain')
        for name in ('source_head','source_tree','source_files','identity_files','binaries','assets','source_manifest','chrome','python','metadata_executables'):
            require(old_manifest[name]==manifest[name],'carried-source-identity')
        for expected in carry['flow_files']:
            original={k:expected[k] for k in ('path','bytes','sha256')}
            require(pin(expected['path'])==original and pin(HERE/expected['name'])['sha256']==expected['sha256'] and old_files[expected['name']]['sha256']==expected['sha256'],'carried-flow-pin')
        bp=Path(approval['browser_receipt'])
        require(str(bp)==carry['browser_receipt']['path'] and pin(bp)['sha256']==approval['browser_receipt_sha256']==carry['browser_receipt']['sha256'],'browser-receipt-pin')
        b=load(bp)
        require(b['mode']=='browser' and b['status']=='PASS' and b['cleanup_complete'] is True and b['browser_dom_observed'] is True and b['seal_sha256']==carry['original_seal']['sha256'] and b['policy_sha256']==carry['original_policy']['sha256'],'browser-prerequisite')
        previous=load(carry['prior_paid_failure']['path'])
        reserved=sum(p['requests_reserved'] for p in previous['observations']['all_proxy_instances'].values())
        require(previous['status']=='FAIL' and previous['cleanup_complete'] is True and reserved==carry['prior_reserved_count']==1 and reserved+policy['total_request_limit']==4,'aggregate-recovery-budget')
    else:
        require(approval['browser_receipt'] is None and approval['browser_receipt_sha256'] is None and approval['browser_accepted'] is False, 'unexpected-browser-authority')
    require(shutil.disk_usage(HERE).free >= 1024**3, 'disk-floor')
    return manifest, policy, approval, policy_sha

class Run:
    def __init__(self, mode, approval_path, run_root, seal):
        self.manifest, self.policy, self.approval, self.policy_sha = preflight(mode, approval_path, run_root, seal)
        self.mode, self.root, self.seal = mode, Path(run_root), seal
        self.root.mkdir(mode=0o700)  # Atomic one-shot reservation, only after all preflight checks.
        self.raw = self.root / 'private'; self.raw.mkdir(mode=0o700)
        for d in ('home', 'tmp', 'work'):
            (self.root/d).mkdir(mode=0o700)
        self.work = self.root/'work'; self.fixture = self.work/'tiny-fixture.txt'
        self.fixture.write_bytes((HERE/'fixture.txt').read_bytes())
        self.fixture_pin = pin(self.fixture)
        self.save = self.work/'conversation.json'
        self.started = time.monotonic(); self.bounds = self.policy['bounds'][mode]
        self.absolute = self.started + self.bounds['absolute_seconds']
        self.active_deadline = self.absolute - 15
        self.failure = None; self.phase = 'setup'; self.processes = []; self.servers = []; self.connections = set(); self.lock = threading.RLock(); self.key_lock = threading.Lock()
        self.secret = None; self.stopping = threading.Event(); self.observations = {}; self.events = []; self.done = threading.Event(); self.browser_dom_observed = False
        self.env = dict(self.policy['environment']); self.env.update(HOME=str(self.root/'home'), TMPDIR=str(self.root/'tmp'))
        AUDIT.authorized = True
        self.watchdog = threading.Thread(target=self._watch, daemon=True); self.watchdog.start()
        self.record('authorized', {'mode': mode, 'seal_sha256': seal, 'policy_sha256': self.policy_sha})
    def record(self, kind, details):
        with self.lock:
            self.events.append({'utc': dt.datetime.now(dt.timezone.utc).isoformat(), 'kind': kind, 'phase': self.phase, 'details': clean(details, self.secret)})
            write_json(self.root/'events.json', self.events, self.secret)
    def check(self, entering=False):
        require(self.failure is None, 'previous-failure')
        now = time.monotonic()
        require(now < min(self.absolute-15, self.active_deadline), 'experiment-deadline')
        if self.phase == 'setup':
            require(now < self.started+self.bounds['setup_seconds'], 'setup-deadline')
        require(dt.datetime.now(dt.timezone.utc) < dt.datetime.fromisoformat(self.approval['expires_utc']), 'approval-expired')
        require(pin(self.fixture) == self.fixture_pin, 'runtime-fixture-changed')
        if entering:
            verify_bundle(self.seal)
            require(shutil.disk_usage(HERE).free >= 1024**3, 'disk-floor')
    def activate(self):
        self.check(); self.phase = 'active'; self.active_deadline = min(self.absolute-15, time.monotonic()+self.bounds['active_seconds'])
    def fail(self, code):
        with self.lock:
            if self.failure is None:
                self.failure = code; self.record('failure', {'code': code})
        self.stopping.set()
    def _watch(self):
        while not self.done.wait(.1):
            try:
                self.check()
            except Refused as error:
                self.fail(str(error)); self.abort_connections(); return
    def abort_connections(self):
        with self.lock:
            for c in list(self.connections):
                try:
                    if c.sock: c.sock.shutdown(socket.SHUT_RDWR)
                except OSError: pass
                try: c.close()
                except OSError: pass
    def child_env(self, origin):
        out = dict(self.env)
        out.update(LLM_VENDOR='anthropic', LLM_MODEL=self.policy['model'], LLM_API_KEY='loopback-placeholder-not-secret', ANTHROPIC_BASE_URL=origin, LLM_SYSTEM=self.policy['system_prompt'], EN_DISABLE_STREAMING='1', ENSEMBLE_DATA_DIR=str(self.work), ENSEMBLE_WORKSPACE=str(self.work), CH02_LOG=str(self.raw/'history.jsonl'))
        require('LLM_BASE_URL' not in out, 'generic-base-url')
        return out
    def start(self, name, argv, env):
        self.check(entering=True)
        stdout = (self.raw/(name+'.stdout')).open('wb'); stderr = (self.raw/(name+'.stderr')).open('wb')
        p = subprocess.Popen(argv, cwd=self.work, env=env, stdin=subprocess.PIPE, stdout=stdout, stderr=stderr, start_new_session=True)
        stdout.close(); stderr.close()
        rec = {'name': name, 'argv': argv, 'environment': env, 'cwd': str(self.work), 'pid': p.pid, 'pgid': p.pid, 'started_utc': dt.datetime.now(dt.timezone.utc).isoformat(), 'signals': [], 'stdout_local': str(self.raw/(name+'.stdout')), 'stderr_local': str(self.raw/(name+'.stderr'))}
        self.processes.append((p,rec)); self.record('spawn', rec); return p,rec
    def end(self, pair, eof=True):
        p,rec = pair
        if eof and p.stdin and not p.stdin.closed: p.stdin.close()
        while p.poll() is None:
            self.check(); time.sleep(.05)
        rec['exit_code'] = p.wait(); rec['ended_utc'] = dt.datetime.now(dt.timezone.utc).isoformat()
        self.record('exit',rec); require(rec['exit_code'] == 0, 'application-exit')
    def metadata(self, argv):
        self.check(entering=True)
        require(argv[0] in ('/bin/ps','/usr/bin/pgrep','/usr/sbin/lsof'), 'metadata-command')
        p = subprocess.run(argv, cwd=self.work, env=self.env, capture_output=True, text=True, timeout=3)
        require(p.returncode in (0,1), 'ownership-query')
        return p.stdout
    def process_tree(self, parent):
        found = {parent}; pending = [parent]
        while pending:
            current = pending.pop()
            for word in self.metadata(['/usr/bin/pgrep','-P',str(current)]).split():
                pid = int(word)
                if pid not in found: found.add(pid); pending.append(pid)
        rows = self.metadata(['/bin/ps','-p',','.join(map(str,sorted(found))),'-o','pid=,ppid=,pgid=,lstart=,command='])
        require(rows.strip(), 'owned-process-absent')
        return found,rows
    def listener(self, pids, port):
        require(port != 8088 and 0 < port < 65536, 'forbidden-listener')
        out = self.metadata(['/usr/sbin/lsof','-nP','-a','-p',','.join(map(str,sorted(pids))),'-iTCP:'+str(port),'-sTCP:LISTEN','-Fpn'])
        require(any(line == 'n127.0.0.1:'+str(port) for line in out.splitlines()), 'listener-not-owned-loopback')
        self.record('listener-owner', {'pids':sorted(pids),'port':port,'lsof':out})
    def credential(self):
        require(self.mode == 'paid-cli', 'paid-mode-required')
        with self.key_lock:
            if self.secret is None:
                self.check(entering=True)
                try:
                    AUDIT.key_window = True
                    with KEY_PATH.open('r', encoding='utf-8') as f:
                        value = json.load(f)['directClaudeAPIKey']
                    require(isinstance(value,str) and bool(value.strip()), 'credential-unavailable')
                    self.secret = value
                except Exception:
                    raise Refused('credential-unavailable') from None
                finally:
                    AUDIT.key_window = False
                self.record('credential-open-audit', {'python_only': True, 'attempts': AUDIT.credential_open_attempts, 'authorized_opens': AUDIT.authorized_credential_opens})
            return self.secret
    def raw_pin_snapshot(self):
        out = []
        for folder in (self.raw, self.work):
            for p in sorted(folder.rglob('*')):
                if p.is_file() and not p.is_symlink():
                    try: out.append(clean(pin(p), self.secret))
                    except OSError: out.append({'path':safe_path(p),'unstable_at_observation':True})
        return out
    def finish(self, success):
        self.phase = 'cleanup'; self.done.set(); self.watchdog.join(1)
        before = self.raw_pin_snapshot()  # observed bytes before cleanup, not a final-stability claim
        self.abort_connections()
        cleanup = True
        for p,rec in reversed(self.processes):
            if p.stdin and not p.stdin.closed:
                try: p.stdin.close()
                except OSError: pass
            if p.poll() is None:
                try: p.wait(timeout=2)
                except subprocess.TimeoutExpired:
                    # Popen still owns this unreaped child identity. No arbitrary PID/group lookup.
                    os.killpg(p.pid, signal.SIGTERM); rec['signals'].append('TERM')
                    try: p.wait(timeout=2)
                    except subprocess.TimeoutExpired:
                        os.killpg(p.pid, signal.SIGKILL); rec['signals'].append('KILL'); p.wait(timeout=2)
            rec['exit_code'] = p.returncode
            until = time.monotonic()+1
            while True:
                try: os.killpg(p.pid,0)
                except ProcessLookupError: rec['group_absent'] = True; break
                if time.monotonic()>=until: rec['group_absent'] = False; cleanup = False; break
                time.sleep(.05)
        for server in reversed(self.servers):
            try: cleanup = server.close() and cleanup
            except Exception: cleanup = False
        after = self.raw_pin_snapshot()
        try: verify_bundle(self.seal)
        except Exception: cleanup = False
        result = {'mode':self.mode,'status':'PASS' if success and not self.failure and cleanup else 'FAIL','failure':self.failure,'source_head':self.manifest['source_head'],'seal_sha256':self.seal,'policy_sha256':self.policy_sha,'run_root':str(self.root),'duration_seconds':time.monotonic()-self.started,'cleanup_complete':cleanup,'processes':self.processes_records(),'observations':self.observations,'browser_dom_observed':self.browser_dom_observed,'human_visual_inspection':False,'credential_audit_python_only':{'attempts':AUDIT.credential_open_attempts,'authorized_opens':AUDIT.authorized_credential_opens},'pre_cleanup_local_hashes_observed_not_final':before,'post_cleanup_local_hashes':after,'scope_limit':'Not host-wide process/network/file-open absence proof; screenshot is not human inspection.'}
        write_json(self.root/'completion.json',result,self.secret); self.secret = None
        return result
    def processes_records(self):
        return [r for _,r in self.processes]

def sequence(steps):
    """Exceptions stop the sequence; no next-phase dispatch/retry/fallback."""
    return [step() for step in steps]
