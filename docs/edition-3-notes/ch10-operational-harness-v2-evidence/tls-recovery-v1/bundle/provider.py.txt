import http.client

class Server:
    def __init__(self, run, handler):
        run.check(entering=True)
        self.run = run; self.closed = False
        outer=self;self.clients=set();self.handlers=set();self.resources_lock=threading.RLock()
        class OwnedHTTP(ThreadingHTTPServer):
            def get_request(inner):
                client,address=super().get_request();client.settimeout(5)
                with outer.resources_lock:outer.clients.add(client)
                return client,address
            def process_request_thread(inner,request,address):
                current=threading.current_thread()
                with outer.resources_lock:outer.handlers.add(current)
                try:super().process_request_thread(request,address)
                finally:
                    with outer.resources_lock:outer.handlers.discard(current);outer.clients.discard(request)
        self.http = OwnedHTTP(('127.0.0.1',0),handler)
        self.http.daemon_threads = True; self.http.block_on_close = False
        self.port = self.http.server_port
        require(self.port != 8088,'forbidden-port')
        self.origin = 'http://127.0.0.1:'+str(self.port)
        self.thread = threading.Thread(target=self.http.serve_forever,kwargs={'poll_interval':.05},daemon=True)
        self.thread.start(); run.servers.append(self)
        run.record('owned-http-listener',{'pid':__import__('os').getpid(),'port':self.port,'origin':self.origin})
    def close(self):
        if self.closed: return True
        self.closed = True; self.http.shutdown()
        with self.resources_lock:
            clients=list(self.clients);handlers=list(self.handlers)
        for client in clients:
            try:client.shutdown(socket.SHUT_RDWR)
            except OSError:pass
            client.close()
        self.http.server_close();self.thread.join(1)
        for handler in handlers:handler.join(.5)
        s = socket.socket(); s.settimeout(.3)
        try: absent = s.connect_ex(('127.0.0.1',self.port)) != 0
        finally: s.close()
        self.run.record('owned-http-closed',{'port':self.port,'thread_joined':not self.thread.is_alive(),'port_refused_at_observation':absent})
        with self.resources_lock:handlers_gone=not self.handlers
        return absent and not self.thread.is_alive() and handlers_gone

def text_values(value):
    if isinstance(value,str): return value
    if isinstance(value,list): return '\n'.join(text_values(x) for x in value)
    if isinstance(value,dict): return '\n'.join(text_values(v) for v in value.values())
    return ''

def blocks(value):
    if isinstance(value,dict):
        if 'type' in value: yield value
        for v in value.values(): yield from blocks(v)
    elif isinstance(value,list):
        for v in value: yield from blocks(v)

class Fixture(Server):
    def __init__(self, run, delayed_headers=False, synthetic=False):
        self.requests = []; self.lock = threading.Lock(); self.delayed = delayed_headers; self.synthetic = synthetic
        outer = self
        class Handler(BaseHTTPRequestHandler):
            def log_message(self,*_): pass
            def do_POST(self):
                try:
                    require(self.path=='/v1/messages','fixture-path')
                    raw=self.rfile.read(int(self.headers['Content-Length'])); body=checked_json(raw)
                    with outer.lock:
                        number=len(outer.requests)+1
                        outer.requests.append({'number':number,'raw_sha256':digest(raw),'body':body})
                    if outer.delayed and number==1:
                        # Delay RESPONSE HEADERS, not body chunks. Overall run deadline still applies.
                        require(not run.stopping.wait(3.5),'fixture-cancelled')
                    if outer.synthetic:
                        content=[{'type':'text','text':'Synthetic boundary-control response.'}]; reason='end_turn'
                    elif number==1:

                    else:
                        results=[b for b in blocks(body.get('messages',[])) if b.get('type')=='tool_result' and b.get('tool_use_id')=='ch10_read_call']
                        require(any('item=cobalt' in text_values(b.get('content')) and 'quantity=7' in text_values(b.get('content')) for b in results),'fixture-needs-actual-read-result')

                    data=json.dumps({'id':'fixture-'+str(number),'type':'message','role':'assistant','model':run.policy['model'],'content':content,'stop_reason':reason,'usage':{'input_tokens':12,'output_tokens':8}}).encode()
                    self.send_response(200);self.send_header('Content-Type','application/json');self.send_header('Content-Length',str(len(data)));self.end_headers();self.wfile.write(data)
                except Exception:
                    run.fail('fake-upstream-failure')

class Proxy(Server):
    def __init__(self, run, native_tokens, fixture=None, guard_case=False):
        require(native_tokens in (1024,4096),'native-token-configuration')
        require(not guard_case or run.mode=='guards','guard-scope')
        require((run.mode=='paid-cli') == (fixture is None),'upstream-kind')
        self.fixture=fixture; self.native_tokens=native_tokens; self.guard_case=guard_case
        self.lock=threading.RLock();self.used=0;self.active=0;self.failed=False;self.records=[];self.requests=[];self.responses=[];self.connections=0
        outer=self
        class Handler(BaseHTTPRequestHandler):
            def log_message(self,*_):pass
            def do_GET(self):self.handle_request()
            def do_PUT(self):self.handle_request()
            def do_DELETE(self):self.handle_request()
            def do_POST(self):self.handle_request()
            def handle_request(self):
                record={'method':self.command,'path':self.path,'reserved':False,'forwarded':False};conn=None
                try:
                    run.check(entering=True)
                    require(self.command=='POST' and self.path=='/v1/messages','exact-method-path')
                    lengths=self.headers.get_all('Content-Length',[])
                    require(len(lengths)==1 and lengths[0].isdigit() and 0<int(lengths[0])<=2*1024*1024 and not self.headers.get('Transfer-Encoding'),'body-length')
                    raw=self.rfile.read(int(lengths[0]));require(len(raw)==int(lengths[0]),'short-body')
                    body=checked_json(raw);require(isinstance(body,dict),'request-object')
                    require(body.get('model')==run.policy['model'],'model')
                    require(type(body.get('max_tokens')) is int and body['max_tokens']==outer.native_tokens,'native-max-tokens')
                    require(body.get('stream',False) is False and isinstance(body.get('messages'),list),'nonstream-request')
                    record.update(body_sha256=digest(raw),model=body['model'],native_max_tokens=body['max_tokens'])
                    with outer.lock:
                        require(not outer.failed,'failed-circuit')
                        require(outer.used<3,'recovery-three-request-cap')
                        outer.used+=1;outer.active+=1;record.update(reserved=True,slot=outer.used)
                    # No refund on failure. Pins/approval checked again before network or credential entry.
                    run.check(entering=True)
                    headers={'Content-Type':'application/json','anthropic-version':self.headers.get('anthropic-version','2023-06-01')}
                    if self.headers.get('anthropic-beta'):headers['anthropic-beta']=self.headers['anthropic-beta']
                    if fixture is None:
                        ca=run.manifest['public_ca'];require(pin(ca['path'])==ca,'public-ca-pin')
                        tls=ssl.create_default_context(cafile=ca['path'])
                        require(tls.check_hostname and tls.verify_mode==ssl.CERT_REQUIRED,'verified-tls-context')
                        record['public_ca_sha256']=ca['sha256'];record['tls_certificate_verification']=True;record['tls_hostname_verification']=True
                        credential=run.credential()  # Only after exact validation, reservation and revalidation.
                        headers['x-api-key']=credential
                        require(credential.encode() not in raw,'credential-in-body')
                        conn=http.client.HTTPSConnection('api.anthropic.com',443,timeout=10,context=tls)
                        endpoint='https://api.anthropic.com/v1/messages'
                    else:
                        headers['x-api-key']='loopback-placeholder-not-secret'
                        conn=http.client.HTTPConnection('127.0.0.1',fixture.port,timeout=10)
                        endpoint=fixture.origin+'/v1/messages'
                    with run.lock:run.connections.add(conn)
                    started=time.monotonic();conn.connect()
                    # A connect timeout is NOT a response/generation timeout. Watchdog owns total bounds.
                    conn.sock.settimeout(None);conn.timeout=None
                    with outer.lock:outer.connections+=1
                    conn.request('POST','/v1/messages',body=raw,headers=headers)
                    response=conn.getresponse();header_elapsed=time.monotonic()-started
                    record.update(endpoint=endpoint,forwarded_body_sha256=digest(raw),upstream_status=response.status,header_elapsed_seconds=header_elapsed)
                    require(not 300<=response.status<400,'redirect-refused')
                    require(response.status==200,'upstream-status')
                    data=response.read(2*1024*1024+1);require(len(data)<=2*1024*1024,'response-size')
                    if run.secret:
                        require(run.secret.encode() not in data and json.dumps(run.secret)[1:-1].encode() not in data,'credential-in-response')
                    parsed=checked_json(data);require(isinstance(parsed,dict) and parsed.get('type')=='message' and isinstance(parsed.get('content'),list),'response-protocol')
                    # Actual tool responses are forwarded unchanged. No preferred tool order or rejection.
                    record.update(response_sha256=digest(data),forwarded=True,usage={k:v for k,v in parsed.get('usage',{}).items() if type(v) is int},usage_kind='provider-measured' if fixture is None else 'fabricated-fixture-usage')
                    with outer.lock:
                        outer.requests.append(body);outer.responses.append(parsed)
                    (run.raw/('request-'+str(record['slot'])+'.json')).write_bytes(raw)
                    (run.raw/('response-'+str(record['slot'])+'.json')).write_bytes(data)
                    self.send_response(200);self.send_header('Content-Type','application/json');self.send_header('Content-Length',str(len(data)));self.end_headers();self.wfile.write(data)
                    record['outcome']='completed'
                except Exception as error:
                    if not isinstance(error,Refused):
                        record['error_type']=type(error).__name__
                        run.record('proxy-transport-error',{'error_type':type(error).__name__,'slot':record.get('slot')})
                    code=str(error) if isinstance(error,Refused) else 'proxy-transport-failure'
                    record.update(outcome='denied-or-failed',failure=code)
                    with outer.lock:outer.failed=True
                    if not outer.guard_case:run.fail(code)
                    try:
                        data=json.dumps({'error':{'type':'harness_failure','message':code}}).encode();self.send_response(429 if code=='recovery-three-request-cap' else 502);self.send_header('Content-Type','application/json');self.send_header('Content-Length',str(len(data)));self.end_headers();self.wfile.write(data)
                    except OSError:pass
                finally:
                    if conn:
                        conn.close()
                        with run.lock:run.connections.discard(conn)
                    with outer.lock:
                        if record['reserved']:outer.active-=1
                        outer.records.append(record)
                    run.record('proxy-request',record)
        super().__init__(run,Handler)
    def facts(self):
        with self.lock:
            return {'requests_reserved':self.used,'active':self.active,'connect_calls':self.connections,'records':list(self.records),'fake_fixture_requests':len(self.fixture.requests) if self.fixture else None,'request_limit_total':3,'audit_limit':'Python proxy/helper observations, not packet capture or host-wide absence proof'}
    def close(self):
        closed=super().close();deadline=time.monotonic()+2
        while self.active and time.monotonic()<deadline:time.sleep(.05)
        self.run.observations.setdefault('all_proxy_instances',{})[self.origin]=self.facts()
        return closed and self.active==0
