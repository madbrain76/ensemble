"""Exactly two application launches per flow; no alternate client campaign."""
import json
import re
import time
from pathlib import Path
from core import HERE, Refused, require, digest, load, pin, sequence
from provider import Fixture, Proxy, blocks, text_values
from wire import Wire

def semantics(text):
    return bool(text.strip()) and 'cobalt' in text.lower() and re.search(r'\b(7|seven)\b',text.lower()) is not None

def archive(run):
    data=load(run.save);require(set(data)=={'config','as_of','context','log'},'archive-envelope')
    require(type(data['as_of']) is int and data['as_of']>0 and isinstance(data['context'],dict) and isinstance(data['log'],list),'archive-types')
    require(data['config']['vendor']=='anthropic' and data['config']['model']==run.policy['model'] and data['config']['system_prompt']==run.policy['system_prompt'],'archive-current-config')
    parts=list(blocks(data));calls={b.get('call_id') for b in parts if b.get('type')=='tool_call' and b.get('name')=='read_file'}
    results={b.get('call_id') for b in parts if b.get('type')=='tool_result' and 'item=cobalt' in text_values(b) and 'quantity=7' in text_values(b)}
    require(bool((calls&results)-{None}),'archive-typed-read-provenance')
    return data

def answer(proxy):
    require(bool(proxy.responses),'missing-provider-response')
    response=proxy.responses[-1]
    require(response.get('stop_reason')=='end_turn','unfinished-turn')
    text='\n'.join(b.get('text','') for b in response.get('content',[]) if b.get('type')=='text')
    require(semantics(text),'answer-semantics')
    return text

def request_provenance(run,proxy):
    calls=[];results=[]
    for body in proxy.requests:
        for b in blocks(body.get('messages',[])):
            if b.get('type')=='tool_use':calls.append(b)
            if b.get('type')=='tool_result':results.append(b)
    reads=[]
    for call in calls:
        if call.get('name')=='read_file':
            p=Path(call.get('input',{}).get('path',''))
            if not p.is_absolute():p=run.work/p
            if p.resolve()==run.fixture.resolve():reads.append(call)
    paired=[(c,r) for c in reads for r in results if r.get('tool_use_id')==c.get('id') and not r.get('is_error',False) and 'item=cobalt' in text_values(r.get('content')) and 'quantity=7' in text_values(r.get('content'))]
    require(bool(paired),'actual-read-tool-result-required')
    return {'paired_read_call_ids':sorted({c['id'] for c,r in paired}),'observed_tool_names':sorted({c.get('name','') for c in calls}),'fact_check':'actual returned item=cobalt and quantity=7; no exact answer phrase or preferred tool order'}

def cards(events):
    result={}
    for e in events:
        kind=e.get('type');part_id=e.get('part_id')
        if part_id is None:continue
        if kind=='part_final':
            c=result.setdefault(str(part_id),{});c.update(part=e.get('part'),role=e.get('role'))
        elif kind in ('tool_dispatched','tool_finished'):
            c=result.setdefault(str(part_id),{});c[kind]={k:v for k,v in e.items() if k not in ('type','replay','at')}
    return result

class CLI:
    def __init__(self,run,proxy):self.run=run;self.proxy=proxy;self.pair=None;self.events=[]
    def start(self,index):
        argv=[self.run.manifest['binaries'][0]['path'],'--save',str(self.run.save),'protocol']
        self.pair=self.run.start('cli-'+str(index),argv,self.run.child_env(self.proxy.origin))
        before=self.proxy.used;until=time.monotonic()+.5
        while time.monotonic()<until:
            self.run.check();require(self.pair[0].poll() is None,'early-cli-exit');time.sleep(.02)
        require(self.proxy.used==before,'startup-provider-request')
        self.run.record('startup-observation',{'application':'cli','index':index,'held_stdin_seconds':.5,'requests_before_user':self.proxy.used,'limit':'bounded observation, not syscall-level proof of all startup work'})
    def prompt(self,text):
        p=self.pair[0];p.stdin.write((json.dumps({'kind':'prompt','text':text})+'\n').encode());p.stdin.flush()
        self.run.end(self.pair,eof=True)
    def stop(self):pass
    def snapshot(self,phase):return {}
    def shutdown(self):pass

class GUI:
    def __init__(self,run,proxy):self.run=run;self.proxy=proxy;self.pair=None;self.wire=None;self.events=[];self.url=None
    def start_process(self,index):
        config_dir=self.run.work/'cr';config_dir.mkdir(exist_ok=True)
        config=config_dir/('settings'+'.json')
        # Newly owned launch configuration only. No theme, TTS, settings or OS preference test.
        if index==1:config.write_text(json.dumps({'tts_enabled':False})+'\n')
        require(load(config).get('tts_enabled') is False,'read-aloud-not-disabled')
        argv=[self.run.manifest['binaries'][1]['path'],'--port','0','--gui-dir',str(Path(self.run.manifest['source_root'])/'gui/web/gui'),'--save',str(self.run.save),'--model',self.run.policy['model'],'--base-url',self.proxy.origin,'--no-stream']
        self.pair=self.run.start('gui-'+str(index),argv,self.run.child_env(self.proxy.origin))
        before=self.proxy.used
        while True:
            self.run.check();require(self.pair[0].poll() is None,'gui-startup-exit')
            error_path=Path(self.pair[1]['stderr_local']);text=error_path.read_text(errors='replace')
            found=re.search(r'GUI http://127\.0\.0\.1:(\d+)',text)
            if found:break
            time.sleep(.05)
        self.port=int(found.group(1));self.run.listener({self.pair[0].pid},self.port);self.url='http://127.0.0.1:'+str(self.port)+'/'
        require(self.proxy.used==before,'startup-provider-request')
        self.run.record('startup-observation',{'application':'gui','index':index,'requests_before_user':self.proxy.used,'listener':self.port})
    def start(self,index):
        self.start_process(index);self.events=[];self.wire=Wire(self.run,self.port)
        self.wire.send({'type':'subscribe'})
        while True:
            e=self.wire.receive();self.events.append(e)
            if e.get('type')=='current_settings':require(e['settings']['tts_enabled'] is False,'read-aloud-not-disabled')
            if e.get('type')=='replay_end':break
    def prompt(self,text):
        self.wire.send({'type':'prompt','text':text})
        while True:
            e=self.wire.receive();self.events.append(e)
            if e.get('type')=='turn_ended':
                require(not e.get('error'),'gui-turn-error');break
    def snapshot(self,phase):
        c=cards(self.events)
        self.run.record('wire-artifacts',{'phase':phase,'cards':c})
        return c
    def stop(self):
        if self.wire:self.wire.close();self.wire=None
        self.run.end(self.pair,eof=True)
    def shutdown(self):pass

def two_flow(run,kind):
    fixture=None if run.mode=='paid-cli' else Fixture(run,delayed_headers=kind=='cli')
    proxy=Proxy(run,1024 if kind=='cli' else 4096,fixture)
    if run.mode=='browser':
        from browser import Browser
        adapter=Browser(run,proxy)
    else:adapter=CLI(run,proxy) if kind=='cli' else GUI(run,proxy)
    state={};phase_facts=[]
    first_prompt=f'Read exactly the tiny local file {run.fixture} using read_file. Do not read other files or modify anything. Report the item and quantity from the actual tool result in one concise sentence. No arbitrary acknowledgement phrase is required.'
    follow_prompt='Using the earlier read_file result already in this conversation, state the item and quantity again without rereading the file. Keep the answer concise.'
    def first():
        adapter.start(1);run.activate();adapter.prompt(first_prompt)
        first_answer=answer(proxy);provenance=request_provenance(run,proxy);before_cards=adapter.snapshot('before-eof')
        if kind=='gui':
            require(any(c.get('part',{}).get('type')=='tool_call' for c in before_cards.values()),'missing-tool-card')
            require(any('tool_finished' in c for c in before_cards.values()),'missing-result-card')
        adapter.stop();saved=archive(run)
        state.update(saved=saved,cards=before_cards,requests=proxy.used)
        phase_facts.append({'phase':'first','answer':first_answer,'archive':pin(run.save),'as_of':saved['as_of'],'requests':proxy.used,'provenance':provenance})
        require(proxy.used<4,'no-resume-budget')
    def second():
        require(load(run.save)==state['saved'],'archive-changed-before-restart')
        adapter.start(2);restored=adapter.snapshot('after-restore-before-user')
        require(proxy.used==state['requests'],'restore-provider-request')
        if kind=='gui':require(restored==state['cards'],'restored-wire-identities-or-payloads')
        adapter.prompt(follow_prompt);second_answer=answer(proxy);after_cards=adapter.snapshot('after-new-final')
        if kind=='gui':
            require(all(after_cards.get(k)==v for k,v in restored.items()),'new-output-overwrote-old-card')
            new={k:v for k,v in after_cards.items() if k not in restored}
            require(any(v.get('part',{}).get('type')=='text' and v.get('role')!='human' for v in new.values()),'new-assistant-final-missing')
        adapter.stop();after=archive(run)
        require(after['as_of']>state['saved']['as_of'] and after['log'][:len(state['saved']['log'])]==state['saved']['log'],'archive-prefix-or-sequence')
        require(proxy.used<=4 and proxy.used>state['requests'],'resume-request-count')
        provenance=request_provenance(run,proxy)
        phase_facts.append({'phase':'resume','answer':second_answer,'archive':pin(run.save),'as_of':after['as_of'],'requests_total':proxy.used,'provenance':provenance,'old_part_ids':sorted(restored),'all_part_ids':sorted(after_cards)})
        if fixture and kind=='cli':
            require(proxy.records[0]['header_elapsed_seconds']>=3.25,'missing-slow-header-observation')
            run.observations['slow_headers']={'actual_elapsed_seconds':proxy.records[0]['header_elapsed_seconds'],'fixture_delay_seconds':3.5,'completed':True}
        run.observations['two_flow']=phase_facts
    try:
        sequence([first,second])
    finally:
        adapter.shutdown()
        run.observations['partial_phases']=phase_facts
        run.observations['proxy']=proxy.facts()
