// All constructors use a valid bootstrap, positive budget, isolated paths and a loopback-only guard.
func reviewConfiguredAgent(t *testing.T,vendor,system string,skills *en.SkillSpec,savePath string) en.Agent {

func reviewAgent(t *testing.T,vendor,system string) en.Agent {t.Helper();return reviewConfiguredAgent(t,vendor,system,nil,"")}
func reviewPrimary(t *testing.T,primary string)(en.Agent,*en.SkillSpec){t.Helper();d:=t.TempDir();p:=filepath.Join(d,"selected","SKILL.md");if e:=os.MkdirAll(filepath.Dir(p),0700);e!=nil{t.Fatal(e)};if e:=os.WriteFile(p,[]byte("---\nname: selected\ntype: primary\ndescription: reviewer selected primary\ntools:\n---\n"+primary),0600);e!=nil{t.Fatal(e)};spec:=&en.SkillSpec{Directory:d,Primary:"selected"};return reviewConfiguredAgent(t,"anthropic","REVIEW_VALID_BOOTSTRAP_NOT_EFFECTIVE",spec,""),spec}
func reviewAppend(t *testing.T,h en.History,e en.Event) { t.Helper();if err:=h.Append(e);err!=nil{t.Fatalf("fixture Append %s: %v",e.Type,err)} }

func TestReviewReplayModesAndRichRender(t *testing.T){
 for _,vendor:=range []string{"anthropic","openai","gemini"}{t.Run(vendor,func(t *testing.T){
  a:=reviewAgent(t,vendor,"PRIMARY\n");reviewRich(t,a,vendor);saved:=reviewSaved(t,a);want:=a.Log().Context();raw:=reviewJSON(t,a.Log().EventsSince(0));var dump bytes.Buffer;if err:=a.Log().Dump(&dump);err!=nil{t.Fatal(err)};render,err:=a.Engine().Render(&want);if err!=nil{t.Fatal(err)}
  t.Logf("canonical context=%s render=%s",reviewJSON(t,want),render)
  for _,mode:=range []string{"checkpoint-only","null-context-full-replay","zero-null-full-replay","zero-cached-plus-tail","ignored-prefix","strict-tail"}{t.Run(mode,func(t *testing.T){
   b:=reviewAgent(t,vendor,"PRIMARY\n");m:=reviewClone(t,saved);expected:=want
   switch mode {
   case "checkpoint-only":m["log"]=json.RawMessage(`[]`)
   case "null-context-full-replay":m["context"]=json.RawMessage(`null`)
   case "zero-null-full-replay":m["context"]=json.RawMessage(`null`);m["as_of"]=json.RawMessage(`0`)
   case "zero-cached-plus-tail":m["as_of"]=json.RawMessage(`0`);base:=en.Part{Type:"text",Text:"BASE_AT_ZERO",From:reviewProv(t,vendor)};tail:=en.Part{Type:"text",Text:"TAIL_AFTER_ZERO",From:reviewProv(t,vendor)};expected.Turn="idle";expected.Dialogue=nil;expected.Ephemera=[]en.Part{base,tail};expected.Usage=en.Usage{Input:987};checkpoint:=expected;checkpoint.Ephemera=[]en.Part{base};m["context"]=reviewJSON(t,checkpoint);m["log"]=reviewJSON(t,[]en.Event{{Seq:1,Type:"hint_received",Message:&en.MessageEvent{Actor:"system",Parts:[]en.Part{tail}}}})
   case "ignored-prefix":var es []en.Event;if e:=json.Unmarshal(m["log"],&es);e!=nil{t.Fatal(e)};es[0].Message.Parts[0].Text="IGNORED_PREFIX";m["log"]=reviewJSON(t,es)
   case "strict-tail":var es []en.Event;if e:=json.Unmarshal(m["log"],&es);e!=nil{t.Fatal(e)};tail:=en.Event{Seq:10,Type:"hint_received",Message:&en.MessageEvent{Actor:"system",Parts:[]en.Part{{Type:"text",Text:"TAIL",From:reviewProv(t,vendor)}}}};es=append(es,tail);m["log"]=reviewJSON(t,es);ref:=reviewAgent(t,vendor,"PRIMARY\n");if e:=ref.Log().Load(bytes.NewReader(dump.Bytes()));e!=nil{t.Fatal(e)};reviewAppend(t,ref.Log(),tail);expected=ref.Log().Context()
   }
   reviewRead(t,b,m);got:=b.Log().Context();if mode=="zero-cached-plus-tail" {if got.Turn!="idle"||len(got.Dialogue)!=0||got.Usage!=expected.Usage||!reflect.DeepEqual(got.Ephemera,expected.Ephemera){t.Fatalf("ZERO_FRONTIER_CHECKPOINT_LOST got=%s want=%s",reviewJSON(t,got),reviewJSON(t,expected))};expected=got};if !reflect.DeepEqual(got,expected){t.Fatalf("PUBLIC_REPLAY_MISMATCH mode=%s\ngot=%s\nwant=%s",mode,reviewJSON(t,got),reviewJSON(t,expected))}
   if !reflect.DeepEqual(b.Log().Context(),expected){t.Fatal("second Context changed replay")};reviewRead(t,b,m);if !reflect.DeepEqual(b.Log().Context(),expected){t.Fatal("repeated ReadSave doubled facts")}
   if mode!="strict-tail"&&mode!="zero-cached-plus-tail"{renderContext:=b.Log().Context();out,e:=b.Engine().Render(&renderContext);if e!=nil{t.Fatal(e)};if !bytes.Equal(out,render){t.Fatalf("render changed across %s",mode)}}
   high:=uint64(9);if mode=="strict-tail"{high=10};if mode=="zero-cached-plus-tail"{high=1};reviewAppend(t,b.Log(),en.Event{Type:"turn_ended",Error:&en.ErrorEvent{}});afterAppend:=b.Log().EventsSince(0);seq:=afterAppend[len(afterAppend)-1].Seq;if seq<=high{t.Fatalf("frontier regressed: %d <= %d",seq,high)}
  })}
  // Public Load must replace a checkpoint base, not append to it.
  b:=reviewAgent(t,vendor,"PRIMARY\n");reviewRead(t,b,saved);reviewAppend(t,b.Log(),en.Event{Type:"hint_received",Message:&en.MessageEvent{Actor:"system",Parts:[]en.Part{{Type:"text",Text:"DISCARD",From:reviewProv(t,vendor)}}}});if e:=b.Log().Load(bytes.NewReader(dump.Bytes()));e!=nil{t.Fatal(e)};if !reflect.DeepEqual(b.Log().Context(),want){t.Fatal("public Load retained checkpoint/stale tail")}
  events:=b.Log().EventsSince(0);events[0].Message.Parts[0].Text="ALIAS";ctx:=b.Log().Context();ctx.Dialogue[0].Parts[0].Text="ALIAS";if !reflect.DeepEqual(b.Log().Context(),want){t.Fatal("public view aliases owner")}
  if !bytes.Equal(reviewJSON(t,b.Log().EventsSince(0)),raw){t.Fatal("archive facts rewritten or aliased")}
 })}
}
func TestReviewMalformedLoadAtomicityAndFrozenPrompt(t *testing.T){
 for _,primary:=range []string{""," \t\n","PRIMARY <literal>\n"}{t.Run(fmt.Sprintf("bytes-%d",len(primary)),func(t *testing.T){a,spec:=reviewPrimary(t,primary);if a.Config().System!=primary{t.Fatal("selected primary not exact")};reviewRich(t,a,"anthropic");before:=reviewJSON(t,a.Log().Context());events:=reviewJSON(t,a.Log().EventsSince(0));m:=reviewSaved(t,a);var cfg map[string]any;if e:=json.Unmarshal(m["config"],&cfg);e!=nil{t.Fatal(e)};if cfg["system_prompt"]!=primary{t.Fatalf("frozen primary changed: %#v",cfg["system_prompt"])}
  snapshotPath:=filepath.Join(t.TempDir(),"primary-snapshot.json");if e:=os.WriteFile(snapshotPath,reviewJSON(t,m),0600);e!=nil{t.Fatal(e)}
  restored:=reviewConfiguredAgent(t,"anthropic","REVIEW_VALID_BOOTSTRAP_NOT_EFFECTIVE",spec,snapshotPath);if restored.Config().System!=primary{t.Fatal("primary fallback after cold startup")};restoredContext:=restored.Log().Context();if !bytes.Equal(reviewJSON(t,restoredContext),before){t.Fatal("primary snapshot context changed")};native,e:=restored.Engine().Render(&restoredContext);if e!=nil{t.Fatal(e)};if bytes.Contains(native,[]byte("REVIEW_VALID_BOOTSTRAP_NOT_EFFECTIVE")){t.Fatal("renderer fell back to bootstrap")};again:=reviewSaved(t,restored);var againConfig en.SaveConfig;if e=json.Unmarshal(again["config"],&againConfig);e!=nil{t.Fatal(e)};if againConfig.SystemPrompt!=primary{t.Fatal("empty/whitespace snapshot was rewritten")}
  current:=reviewAgent(t,"anthropic","CURRENT_CONFIGURATION_WINS");if e:=current.Load(snapshotPath);e!=nil{t.Fatal(e)};if current.Config().System!="CURRENT_CONFIGURATION_WINS"{t.Fatal("recorded metadata replaced active instructions")};t.Logf("valid selected and persisted primary bytes=%d; cold render has no bootstrap fallback; current configuration preserved",len(primary))
  for _,bad:=range []string{`{}`,`{"config":null,"context":null,"as_of":0,"log":[]}`,strings.Replace(string(reviewJSON(t,m)),`"turn_ended"`,`"UNKNOWN_EVENT"`,1),string(reviewJSON(t,m))+` {}`} {if e:=a.Log().ReadSave(strings.NewReader(bad));e==nil{t.Fatalf("accepted malformed save %q",bad)};if !bytes.Equal(reviewJSON(t,a.Log().Context()),before)||!bytes.Equal(reviewJSON(t,a.Log().EventsSince(0)),events){t.Fatal("ReadSave failure changed visible owner state")}}
  path:=filepath.Join(t.TempDir(),"bad.json");if e:=os.WriteFile(path,[]byte(`{}`),0600);e!=nil{t.Fatal(e)};if e:=a.Load(path);e==nil{t.Fatal("public Load accepted malformed save")};if !bytes.Equal(reviewJSON(t,a.Log().Context()),before)||!bytes.Equal(reviewJSON(t,a.Log().EventsSince(0)),events){t.Fatal("Agent.Load failure changed state")};if a.Config().System!=primary{t.Fatal("failed Load replaced current primary")}
 })}
}
