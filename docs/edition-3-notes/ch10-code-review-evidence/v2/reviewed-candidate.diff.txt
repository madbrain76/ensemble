+# Conversation save/restore
+
+The CLI (`cmd/ch01`) and GUI (`gui/cmd`) accept `--save PATH`, defaulting to
+working-directory `save.json`. Missing means fresh. Any other load failure
+refuses startup without replacing the input. EOF saves after admitted requests
+finish; GUI signal shutdown cancels/drains its active turn and checkpoints.
+Use separate files for independent client instances; concurrent writers to one
+file are not supported.
+
+Library Agents load on construction and save on `Ensemble.Shutdown`. Their
+archive defaults to `filepath.Join(Config.DataDir, "save.json")`. An unspecified
+DataDir receives a unique temporary Agent directory (exposed by `Config()`);
+pass a stable explicit directory to resume across root instances. The caller owns
+that directory's lifetime. `Config.SavePath` overrides only the archive location.
+`Agent.Save(path...)` / `Agent.Load(path...)` serialize through the Agent owner;
+loads refuse active/queued work, and omitted paths select the lifecycle location.
+An explicit method argument does not change that lifecycle location. Shutdown
+returns save errors. Parent directories are created with private permissions;
+serialization, sync and close complete in a sibling temporary file before rename.
+This protects the previous archive from serialization/write failure, not from all
+power-loss or hostile filesystem scenarios.
+
+The four-field JSON envelope is exported as `SaveFile`: `config`, `as_of`,
+`context`, `log`. Informational `SaveConfig` contains model, vendor,
+`system_prompt`, and plain tool declarations (`name`, `description`, `schema`).
+It reflects the running Agent's effective primary-skill instruction and tool
+surface; it never reinstalls runtime configuration, credentials, executable tools,
+skill activation, queues, jobs or workers. Credentials/base URLs/workspace and
+process handles are not archive config. Ordinary conversation contents may of
+course include sensitive material supplied by the application.
+
+History owns validation and checkpoint reduction. With a non-null context it
+copies that checkpoint and applies only retained events with sequence greater
+than `as_of`. A null context folds the retained log from zero. Every retained
+fact—including the older prefix—is validated before installation. New sequences
+start beyond both checkpoint and retained log; later saves retain the effective
+checkpoint even when the log prefix was trimmed. Dialogue, pending ephemera,
+usage, turn state, rich parts, provenance and original event payloads survive.
+`History.Context()` remains detached. Existing JSONL `Dump`/`Load` and CLI
+`dump`/`render` are separate offline compatibility/inspection paths, not the
+lifecycle archive format.
+
+`Ephemeral{Text: ...}` is an explicit public mailbox input for durable system
+context without starting an idle turn. Existing `Hint` behavior is unchanged.
+The CLI's strict envelopes, rich `parts` records and live protocol remain
+supported. A trailing CLI ephemeral envelope is persisted even without a prompt.
+
+Local regression tests are `go test ./...` (root) and `go test ./...` in `gui`.
+Chapter10 tests use local scripted providers only; they are not browser/native
+speech or real-provider evidence. Actual browser/provider exercise requires the
+separate authorization and evidence gate.

+func archiveAgent(t *testing.T, cfg ensemble.Config) ensemble.Agent {
+	t.Helper()
+	if cfg.System == "" && cfg.Skills == nil {
+		cfg.System = "current instruction"
+	}
+	if cfg.MaxTokens == 0 {
+		cfg.MaxTokens = 128
+	}
+	if cfg.Model == "" {
+		cfg.Model = "exact-model"
+	}
+	if cfg.DataDir == "" {
+		cfg.DataDir = t.TempDir()
+	}
+	r := ensemble.New(log.New(io.Discard, "", 0))
+	a, err := r.NewAgent(cfg)
+	if err != nil {
+		t.Fatal(err)
+	}
+	t.Cleanup(func() {
+		if err := r.Shutdown(); err != nil {
+			t.Error(err)
+		}
+	})
+	return a
+}
+func writeArchiveFixture(t *testing.T, s ensemble.SaveFile) string {
+	t.Helper()
+	path := filepath.Join(t.TempDir(), "save.json")
+	b, err := json.Marshal(s)
+	if err != nil {
+		t.Fatal(err)
+	}
+	if err := os.WriteFile(path, b, 0600); err != nil {
+		t.Fatal(err)
+	}
+	return path
+}
+func readArchiveFixture(t *testing.T, path string) ensemble.SaveFile {
+	t.Helper()
+	b, err := os.ReadFile(path)
+	if err != nil {
+		t.Fatal(err)
+	}
+	var s ensemble.SaveFile
+	if err := json.Unmarshal(b, &s); err != nil {
+		t.Fatal(err)
+	}
+	return s

+func sameJSON(t *testing.T, a, b any) {

+	y, _ := json.Marshal(b)
+	if !bytes.Equal(x, y) {
+		t.Fatalf("different\n%s\n%s", x, y)
+	}

+func TestChapter10ValidationIsAtomic(t *testing.T) {
+	a := archiveAgent(t, ensemble.Config{})
+	appendFact(t, a, user("existing"))
+	original := a.Log().Context()
+	originalEvents := a.Log().Events()
+	path := filepath.Join(t.TempDir(), "base.json")
+	if err := a.Save(path); err != nil {
+		t.Fatal(err)
+	}
+	good, _ := os.ReadFile(path)
+	cases := map[string]func(map[string]any){
+		"missing config": func(m map[string]any) { delete(m, "config") },
+		"bad config":     func(m map[string]any) { m["config"].(map[string]any)["tools"] = "not tools" },
+		"missing prompt": func(m map[string]any) { delete(m["config"].(map[string]any), "system_prompt") },
+		"bad schema": func(m map[string]any) {
+			m["config"].(map[string]any)["tools"] = []any{map[string]any{"name": "x", "description": "", "schema": nil}}
+		},
+		"missing context": func(m map[string]any) { delete(m, "context") },
+		"bad state":       func(m map[string]any) { m["context"].(map[string]any)["turn"] = "garbage" },
+		"negative usage":  func(m map[string]any) { m["context"].(map[string]any)["usage"].(map[string]any)["input"] = -1 },
+		"late invalid event": func(m map[string]any) {
+			m["log"] = append(m["log"].([]any), map[string]any{"seq": 2, "type": "unknown", "message": map[string]any{"actor": "human"}})
+		},
+		"bad old prefix":     func(m map[string]any) { m["log"].([]any)[0].(map[string]any)["type"] = "unknown" },
+		"duplicate sequence": func(m map[string]any) { m["log"] = append(m["log"].([]any), m["log"].([]any)[0]) },
+		"future context":     func(m map[string]any) { m["as_of"] = 0 },
+	}
+	for name, mutate := range cases {
+		t.Run(name, func(t *testing.T) {
+			var m map[string]any
+			_ = json.Unmarshal(good, &m)
+			mutate(m)
+			bad, _ := json.Marshal(m)
+			file := filepath.Join(t.TempDir(), "bad.json")
+			_ = os.WriteFile(file, bad, 0600)
+			if err := a.Load(file); err == nil {
+				t.Fatal("accepted malformed archive")
+			}
+			sameJSON(t, a.Log().Context(), original)
+			sameJSON(t, a.Log().Events(), originalEvents)
+			unchanged, _ := os.ReadFile(file)
+			if !bytes.Equal(bad, unchanged) {
+				t.Fatal("damaged archive overwritten")
+			}
+			root := ensemble.New(log.New(io.Discard, "", 0))
+			if _, err := root.NewAgent(ensemble.Config{System: "runtime", Model: "m", MaxTokens: 1, SavePath: file, DataDir: t.TempDir()}); err == nil {
+				t.Fatal("startup accepted malformed save")
+			}
+			if err := root.Shutdown(); err != nil {
+				t.Fatal(err)
+			}
+			unchanged, _ = os.ReadFile(file)
+			if !bytes.Equal(bad, unchanged) {
+				t.Fatal("failed startup overwrote archive")
+			}
+		})
+	}
+}

+		return fmt.Errorf("missing context")
+	}
+	if err := required(fields["config"], "model", "vendor", "system_prompt", "tools"); err != nil {
+		return fmt.Errorf("config: %w", err)
+	}
+	var save common.SaveFile
+	d := json.NewDecoder(bytes.NewReader(data))
+	d.DisallowUnknownFields()
+	if err := d.Decode(&save); err != nil {
+		return err
+	}
+	if d.Decode(new(any)) != io.EOF {
+		return fmt.Errorf("trailing save data")
+	}
+	if save.Config.Vendor < common.Anthropic || save.Config.Vendor > common.Gemini {
+		return fmt.Errorf("invalid recorded model/vendor")
+	}
+	if save.AsOf == math.MaxUint64 {
+		return fmt.Errorf("sequence exhausted")
+	}
+	names := map[string]bool{}
+	for _, tool := range save.Config.Tools {
+		var schema map[string]json.RawMessage
+		if tool.Name == "" || names[tool.Name] || json.Unmarshal(tool.Schema, &schema) != nil || schema == nil {
+			return fmt.Errorf("invalid tool declaration")
+		}
+		names[tool.Name] = true
+	}
+	// Require explicit declaration fields even when their strings are empty.
+	var cfg struct {
+		Tools []json.RawMessage `json:"tools"`
+	}
+	if err := json.Unmarshal(fields["config"], &cfg); err != nil {
+		return err
+	}
+	for _, tool := range cfg.Tools {
+		if err := required(tool, "name", "description", "schema"); err != nil {
+			return err
+		}
+	}
+	if save.Context != nil {
+		if err := required(fields["context"], "turn", "usage"); err != nil {
+			return err
+		}
+		var cf map[string]json.RawMessage
+		_ = json.Unmarshal(fields["context"], &cf)
+		for _, k := range []string{"dialogue", "ephemera"} {
+			if _, ok := cf[k]; !ok {
+				return fmt.Errorf("missing context %s", k)
+			}
+		}
+		if err := validContext(*save.Context, save.AsOf); err != nil {
+			return err
+		}
+	}
+	var previous uint64
+	frontier := save.AsOf
+	for _, e := range save.Log {
+		if e.Seq == 0 || e.Seq <= previous || e.Seq == math.MaxUint64 {
+			return fmt.Errorf("log sequence must increase and leave room")
+		}
+		if err := validate(e); err != nil {
+			return fmt.Errorf("event %d: %w", e.Seq, err)
+		}
+		previous = e.Seq
+		if e.Seq > frontier {
+			frontier = e.Seq
+		}
+	}
+	h.mu.Lock()
+	defer h.mu.Unlock()
+	h.events, h.base, h.anchor, h.frontier = save.Log, save.Context, save.AsOf, frontier
+	return nil
+}
+func validContext(c common.Context, anchor uint64) error {
+	switch c.Turn {
+	case common.Idle, common.InputPending, common.InFlight, common.ToolsPending, common.Interrupted:
+	default:
+		return fmt.Errorf("invalid context turn")
+	}
+	if c.Usage.Input < 0 || c.Usage.Output < 0 || c.Usage.CacheRead < 0 || c.Usage.CacheWrite < 0 {
+		return fmt.Errorf("negative context usage")
+	}
+	var previous uint64
+	for _, t := range c.Dialogue {
+		if t.Seq == 0 || t.Seq <= previous || t.Seq > anchor {
+			return fmt.Errorf("invalid context sequence")
+		}
+		previous = t.Seq
+		switch t.Actor {
+		case "human", "agent", "tool", "system":
+		default:
+			return fmt.Errorf("invalid context actor")
+		}
+		if err := partsValid(t.Parts); err != nil {
+			return err
+		}
+	}
+	return partsValid(c.Ephemera)
+}
