func reviewProv(t *testing.T, vendor string) en.Provenance {
 t.Helper(); model:=map[string]string{"anthropic":"claude-sonnet-4-6","openai":"gpt-5-course","gemini":"gemini-2.5-pro"}[vendor]
 surface:=map[string]string{"anthropic":"messages","openai":"chat_completions","gemini":"generate_content"}[vendor]
 var p en.Provenance; b,_:=json.Marshal(map[string]string{"vendor":vendor,"model":model,"surface":surface});if err:=json.Unmarshal(b,&p);err!=nil{t.Fatal(err)};return p
}
func reviewAgent(t *testing.T,vendor,system string) en.Agent {
 t.Helper();p:=reviewProv(t,vendor);r:=en.New(log.New(io.Discard,"",0));d:=t.TempDir();a,err:=r.NewAgent(en.Config{Vendor:p.Vendor,Model:p.Model,ResolvedModel:p.Model,System:system,DataDir:d,SavePath:filepath.Join(d,"save.json"),DisableStreaming:true});if err!=nil{t.Fatal(err)};t.Cleanup(func(){if err:=r.Shutdown();err!=nil{t.Errorf("cleanup: %v",err)}});return a
}
func reviewJSON(t *testing.T,v any) []byte {t.Helper();b,e:=json.Marshal(v);if e!=nil{t.Fatal(e)};return b}
func reviewSaved(t *testing.T,a en.Agent) map[string]json.RawMessage {t.Helper();p:=filepath.Join(t.TempDir(),"archive.json");if e:=a.Save(p);e!=nil{t.Fatal(e)};b,e:=os.ReadFile(p);if e!=nil{t.Fatal(e)};var m map[string]json.RawMessage;if e=json.Unmarshal(b,&m);e!=nil{t.Fatal(e)};return m}
func reviewRead(t *testing.T,a en.Agent,m map[string]json.RawMessage){t.Helper();if e:=a.Log().ReadSave(bytes.NewReader(reviewJSON(t,m)));e!=nil{t.Fatal(e)}}
func reviewClone(t *testing.T,m map[string]json.RawMessage) map[string]json.RawMessage {t.Helper();var n map[string]json.RawMessage;if e:=json.Unmarshal(reviewJSON(t,m),&n);e!=nil{t.Fatal(e)};return n}
func reviewRich(t *testing.T,a en.Agent,vendor string) {
 t.Helper();p:=reviewProv(t,vendor);text:=func(s string)en.Part{return en.Part{Type:"text",Text:s,From:p}}
 opaque:=map[string]string{"anthropic":`{"type":"thinking","thinking":"SYNTHETIC_THOUGHT","signature":"SYNTHETIC_SIG"}`,"openai":`{"reasoning_content":"SYNTHETIC_THOUGHT"}`,"gemini":`{"thought":true,"text":"SYNTHETIC_THOUGHT","thoughtSignature":"SYNTHETIC_SIG"}`}[vendor]
 callOpaque:=map[string]string{"anthropic":`{"type":"tool_use","id":"call-one","name":"noop","input":{"x":1}}`,"openai":`{"id":"call-one","type":"function","function":{"name":"noop","arguments":"{\"x\":1}"}}`,"gemini":`"SYNTHETIC_SIG"`}[vendor]
 h:=a.Log();h.Append(en.Event{Type:"message_received",Message:&en.MessageEvent{Actor:"human",Parts:[]en.Part{text("HUMAN"),text("SECOND_PART")}}})
 h.Append(en.Event{Type:"response_ended",Response:&en.ResponseEvent{From:p,Usage:en.Usage{Input:3,Output:4},Parts:[]en.Part{text("AGENT"),{Type:"opaque",From:p,Data:json.RawMessage(opaque)},{Type:"blob",MIME:"image/png",Ref:&en.Ref{Kind:1,Locator:"synthetic-image.png"},From:p},{Type:"tool_call",CallID:"call-one",Name:"noop",Args:json.RawMessage(`{"x":1}`),Opaque:json.RawMessage(callOpaque),From:p}}}})
 h.Append(en.Event{Type:"tool_called",Tool:&en.ToolEvent{CallID:"call-one",Name:"noop",Args:json.RawMessage(`{"x":1}`),Dispatched:true,DisplayID:31}})
 h.Append(en.Event{Type:"tool_returned",Tool:&en.ToolEvent{CallID:"call-one",Name:"noop",Parts:[]en.Part{text("RESULT"),{Type:"blob",MIME:"application/pdf",Ref:&en.Ref{Kind:1,Locator:"synthetic-result.pdf"},From:p}},DisplayID:31}})
 h.Append(en.Event{Type:"hint_received",Message:&en.MessageEvent{Actor:"system",Parts:[]en.Part{text("PENDING_HINT")}}})
 h.Append(en.Event{Type:"response_ended",Response:&en.ResponseEvent{From:p,Usage:en.Usage{Input:2,Output:1},Parts:[]en.Part{text("TO_SUMMARIZE")}}})

}
func TestReviewReplayModesAndRichRender(t *testing.T){
 for _,vendor:=range []string{"anthropic","openai","gemini"}{t.Run(vendor,func(t *testing.T){
  a:=reviewAgent(t,vendor,"PRIMARY\n");reviewRich(t,a,vendor);saved:=reviewSaved(t,a);want:=a.Log().Context();raw:=reviewJSON(t,a.Log().EventsSince(0));render,err:=a.Engine().Render(want);if err!=nil{t.Fatal(err)}
  t.Logf("canonical context=%s render=%s",reviewJSON(t,want),render)
  for _,mode:=range []string{"checkpoint-only","null-context-full-replay","zero-null-full-replay","zero-cached-full-replay","ignored-prefix","strict-tail"}{t.Run(mode,func(t *testing.T){
   b:=reviewAgent(t,vendor,"PRIMARY\n");m:=reviewClone(t,saved);expected:=want
   switch mode {
   case "checkpoint-only":m["log"]=json.RawMessage(`[]`)
   case "null-context-full-replay":m["context"]=json.RawMessage(`null`)
   case "zero-null-full-replay":m["context"]=json.RawMessage(`null`);m["as_of"]=json.RawMessage(`0`)
   case "zero-cached-full-replay":m["as_of"]=json.RawMessage(`0`);m["context"]=json.RawMessage(`{"turn":"Idle","dialogue":[],"ephemera":[{"actor":"system","parts":[{"type":"text","text":"STALE_CACHE","from":{"vendor":"anthropic","model":"irrelevant","surface":"messages"}}],"seq":0}],"usage":{"input":987,"cache_write":0,"cache_read":0,"output":0}}`)
   case "ignored-prefix":var es []en.Event;if e:=json.Unmarshal(m["log"],&es);e!=nil{t.Fatal(e)};es[0].Message.Parts[0].Text="IGNORED_PREFIX";m["log"]=reviewJSON(t,es)
   case "strict-tail":var es []en.Event;if e:=json.Unmarshal(m["log"],&es);e!=nil{t.Fatal(e)};tail:=en.Event{Seq:10,Type:"hint_received",Message:&en.MessageEvent{Actor:"system",Parts:[]en.Part{{Type:"text",Text:"TAIL",From:reviewProv(t,vendor)}}}};es=append(es,tail);m["log"]=reviewJSON(t,es);ref:=reviewAgent(t,vendor,"PRIMARY\n");if e:=ref.Log().Load(bytes.NewReader(raw));e!=nil{t.Fatal(e)};ref.Log().Append(tail);expected=ref.Log().Context()
   }
   reviewRead(t,b,m);got:=b.Log().Context();if !reflect.DeepEqual(got,expected){t.Fatalf("PUBLIC_REPLAY_MISMATCH mode=%s\ngot=%s\nwant=%s",mode,reviewJSON(t,got),reviewJSON(t,expected))}
   if !reflect.DeepEqual(b.Log().Context(),expected){t.Fatal("second Context changed replay")};reviewRead(t,b,m);if !reflect.DeepEqual(b.Log().Context(),expected){t.Fatal("repeated ReadSave doubled facts")}
   if mode!="strict-tail"{out,e:=b.Engine().Render(b.Log().Context());if e!=nil{t.Fatal(e)};if !bytes.Equal(out,render){t.Fatalf("render changed across %s",mode)}}
   high:=uint64(9);if mode=="strict-tail"{high=10};if seq:=b.Log().Append(en.Event{Type:"turn_ended"});seq<=high{t.Fatalf("frontier regressed: %d <= %d",seq,high)}
  })}
  // Public Load must replace a checkpoint base, not append to it.
  b:=reviewAgent(t,vendor,"PRIMARY\n");reviewRead(t,b,saved);b.Log().Append(en.Event{Type:"hint_received",Message:&en.MessageEvent{Actor:"system",Parts:[]en.Part{{Type:"text",Text:"DISCARD",From:reviewProv(t,vendor)}}}});if e:=b.Log().Load(bytes.NewReader(raw));e!=nil{t.Fatal(e)};if !reflect.DeepEqual(b.Log().Context(),want){t.Fatal("public Load retained checkpoint/stale tail")}
  events:=b.Log().EventsSince(0);events[0].Message.Parts[0].Text="ALIAS";ctx:=b.Log().Context();ctx.Dialogue[0].Parts[0].Text="ALIAS";if !reflect.DeepEqual(b.Log().Context(),want){t.Fatal("public view aliases owner")}
  if !bytes.Equal(reviewJSON(t,b.Log().EventsSince(0)),raw){t.Fatal("archive facts rewritten or aliased")}
 })}
}
func TestReviewMalformedLoadAtomicityAndFrozenPrompt(t *testing.T){
 for _,primary:=range []string{""," \t\n","PRIMARY <literal>\n"}{t.Run(fmt.Sprintf("bytes-%d",len(primary)),func(t *testing.T){a:=reviewAgent(t,"anthropic",primary);reviewRich(t,a,"anthropic");before:=reviewJSON(t,a.Log().Context());events:=reviewJSON(t,a.Log().EventsSince(0));m:=reviewSaved(t,a);var cfg map[string]any;if e:=json.Unmarshal(m["config"],&cfg);e!=nil{t.Fatal(e)};if cfg["system_prompt"]!=primary{t.Fatalf("frozen primary changed: %#v",cfg["system_prompt"])}
  for _,bad:=range []string{`{}`,`{"config":null,"context":null,"as_of":0,"log":[]}`,strings.Replace(string(reviewJSON(t,m)),`"turn_ended"`,`"UNKNOWN_EVENT"`,1),string(reviewJSON(t,m))+` {}`} {if e:=a.Log().ReadSave(strings.NewReader(bad));e==nil{t.Fatalf("accepted malformed save %q",bad)};if !bytes.Equal(reviewJSON(t,a.Log().Context()),before)||!bytes.Equal(reviewJSON(t,a.Log().EventsSince(0)),events){t.Fatal("ReadSave failure changed visible owner state")}}
  path:=filepath.Join(t.TempDir(),"bad.json");if e:=os.WriteFile(path,[]byte(`{}`),0600);e!=nil{t.Fatal(e)};if e:=a.Load(path);e==nil{t.Fatal("public Load accepted malformed save")};if !bytes.Equal(reviewJSON(t,a.Log().Context()),before)||!bytes.Equal(reviewJSON(t,a.Log().EventsSince(0)),events){t.Fatal("Agent.Load failure changed state")};if a.Config().System!=primary{t.Fatal("failed Load replaced current primary")}
 })}
}
func TestReviewAtomicSavePreservesFile(t *testing.T){
 if os.Geteuid()==0{t.Fatal("permission oracle requires non-root identity")};a:=reviewAgent(t,"anthropic","PRIMARY");a.Log().Append(en.Event{Type:"turn_ended"});dir:=t.TempDir();path:=filepath.Join(dir,"sentinel.json");sentinel:=[]byte("REVIEW_ORIGINAL_REGULAR_FILE\n");if e:=os.WriteFile(path,sentinel,0600);e!=nil{t.Fatal(e)};if e:=os.Chmod(dir,0500);e!=nil{t.Fatal(e)};defer os.Chmod(dir,0700)
 err:=a.Save(path);if err==nil{t.Fatal("controlled read-only parent did not reject Save")};got,e:=os.ReadFile(path);if e!=nil{t.Fatal(e)};if !bytes.Equal(got,sentinel){t.Fatalf("ATOMIC_SENTINEL_CORRUPTED: got=%q want=%q; Save error=%v",got,sentinel,err)};t.Logf("actual save error=%v; sentinel unchanged",err)
}
