type reviewLockedBuffer struct {sync.Mutex;b bytes.Buffer}
func (b *reviewLockedBuffer) Write(p []byte)(int,error){b.Lock();defer b.Unlock();return b.b.Write(p)}
func (b *reviewLockedBuffer) text()string{b.Lock();defer b.Unlock();return b.b.String()}
type reviewSocket struct{net.Conn;r *bufio.Reader}
func reviewWS(t *testing.T,origin string)*reviewSocket{t.Helper();u,e:=url.Parse(origin);if e!=nil{t.Fatal(e)};c,e:=net.DialTimeout("tcp",u.Host,3*time.Second);if e!=nil{t.Fatal(e)};fmt.Fprintf(c,"GET /ws HTTP/1.1\r\nHost: %s\r\nUpgrade: websocket\r\nConnection: Upgrade\r\nSec-WebSocket-Version: 13\r\nSec-WebSocket-Key: cmV2aWV3LW93bmVkLXdzIQ==\r\n\r\n",u.Host);r:=bufio.NewReader(c);line,e:=r.ReadString('\n');if e!=nil||!strings.Contains(line,"101"){c.Close();t.Fatalf("upgrade: %q %v",line,e)};for{line,e=r.ReadString('\n');if e!=nil{t.Fatal(e)};if line=="\r\n"{break}};return &reviewSocket{Conn:c,r:r}}

func(s *reviewSocket)read(t *testing.T)map[string]any{t.Helper();s.SetReadDeadline(time.Now().Add(6*time.Second));h:=make([]byte,2);if _,e:=io.ReadFull(s.r,h);e!=nil{t.Fatal(e)};n:=uint64(h[1]&127);if n==126{b:=make([]byte,2);if _,e:=io.ReadFull(s.r,b);e!=nil{t.Fatal(e)};n=uint64(binary.BigEndian.Uint16(b))};if n==127{b:=make([]byte,8);if _,e:=io.ReadFull(s.r,b);e!=nil{t.Fatal(e)};n=binary.BigEndian.Uint64(b)};if h[0]&15!=1||n>1<<20||h[1]&128!=0{t.Fatalf("unexpected frame %v/%d",h,n)};b:=make([]byte,int(n));if _,e:=io.ReadFull(s.r,b);e!=nil{t.Fatal(e)};var m map[string]any;if e:=json.Unmarshal(b,&m);e!=nil{t.Fatal(e)};return m}

func reviewID(v any)string{b,_:=json.Marshal(v);return string(b)}

}
