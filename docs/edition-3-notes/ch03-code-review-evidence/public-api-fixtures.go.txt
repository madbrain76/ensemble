// All text/signatures in these independent fixtures are synthetic nonsecrets.
func reviewAgent(t *testing.T, url string) ensemble.Agent {
 t.Helper()
 a,err:=ensemble.New(log.New(io.Discard,"",0)).NewAgent(ensemble.Config{Vendor:ensemble.Gemini,Model:"gemini-3.8-flash",APIKey:"fixture-only",System:"Use tools carefully.",MaxTokens:1000,BaseURL:url,EnableTools:true,RequireToolExplanation:true})
 if err!=nil {t.Fatal(err)}; return a
}
func reviewReply(parts []any) []byte {
 b,_:=json.Marshal(map[string]any{"modelVersion":"gemini-3.8-flash","candidates":[]any{map[string]any{"content":map[string]any{"role":"model","parts":parts}}},"usageMetadata":map[string]int{"promptTokenCount":7,"candidatesTokenCount":5}});return b
}
func TestReviewGeminiSignedOrdinaryText(t *testing.T) {
 for _,thought:=range []string{"absent","false"} {t.Run(thought,func(t *testing.T){
  a:=reviewAgent(t,""); part:=map[string]any{"text":"Public completion.","thoughtSignature":"synthetic-public-signature"}; if thought=="false" {part["thought"]=false}
  raw:=reviewReply([]any{part,map[string]any{"text":"synthetic-hidden-reasoning","thought":true,"thoughtSignature":"synthetic-hidden-signature"}})
  event,err:=a.Engine().Parse(raw,a.Engine().Target());if err!=nil {t.Fatal(err)}
  visible:="";for _,p:=range event.Parts {if p.Type=="text" {visible+=p.Text}}
  if visible!="Public completion." {t.Errorf("visible ordinary text=%q; signature alone must not classify text as hidden",visible)}
  if err:=a.Log().Append(ensemble.Event{Type:"response_ended",Response:&event});err!=nil{t.Fatal(err)}
  var dump bytes.Buffer;if err:=a.Log().Dump(&dump);err!=nil{t.Fatal(err)}; loaded:=reviewAgent(t,"");if err:=loaded.Log().Load(&dump);err!=nil{t.Fatal(err)}
  same,err:=loaded.Engine().Render(loaded.Log().Context(),loaded.Engine().Target());if err!=nil{t.Fatal(err)}
  if !bytes.Contains(same,[]byte("synthetic-public-signature")) || !bytes.Contains(same,[]byte("Public completion.")) || !bytes.Contains(same,[]byte("synthetic-hidden-signature")){t.Error("same-model replay lost source-bound material")}
  for _,target:=range []ensemble.Provenance{{Vendor:ensemble.Gemini,Surface:ensemble.GenerateContent,Model:"different-model"},{Vendor:ensemble.Anthropic,Surface:ensemble.Messages,Model:"different-model"},{Vendor:ensemble.OpenAI,Surface:ensemble.ChatCompletions,Model:"different-model"}} {
   cross,err:=loaded.Engine().Render(loaded.Log().Context(),target);if err!=nil{t.Fatal(err)}
   if !bytes.Contains(cross,[]byte("Public completion.")){t.Errorf("ordinary text lost crossing to %v",target)}
   if bytes.Contains(cross,[]byte("synthetic-")){t.Errorf("hidden/signature leaked crossing to %v",target)}
  }
 })}
}
func TestReviewGeminiSignedExplanationAndAnswer(t *testing.T){
 marker:=filepath.Join(t.TempDir(),"marker");round:=0;refused:=false
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  body,_:=io.ReadAll(r.Body);round++;refused=refused||bytes.Contains(body,[]byte("tool batch refused"))
  if round==1 {_,_=w.Write(reviewReply([]any{map[string]any{"text":"I will write the requested marker.","thoughtSignature":"synthetic-explanation-signature"},map[string]any{"functionCall":map[string]any{"name":"write_file","args":map[string]string{"path":marker,"content":"done"}},"thoughtSignature":"synthetic-call-signature"}}));return}
  _,_=w.Write(reviewReply([]any{map[string]any{"text":"The marker is complete.","thoughtSignature":"synthetic-answer-signature"}}))
 }));defer server.Close()
 a:=reviewAgent(t,server.URL);answer,err:=a.Ask(context.Background(),"write a marker")
 if err!=nil{t.Error(err)};if answer!="The marker is complete." {t.Errorf("signed ordinary terminal answer=%q",answer)}
 if _,err:=os.Stat(marker);err!=nil {t.Errorf("valid signed explanation did not allow tool: %v",err)}
 if refused {t.Error("signed ordinary explanation was refused")};if round!=2{t.Errorf("requests=%d",round)}
 if u:=a.Engine().Usage();u.Input!=14||u.Output!=10{t.Errorf("usage=%+v",u)}
 // Hidden reasoning is never expected in a public answer.
 if strings.Contains(answer,"synthetic-"){t.Error("private fixture material in answer")}
}

func TestReviewDistinctFallbackIDs(t *testing.T){
 round:=0
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  round++
  if round==1 {_,_=w.Write(reviewReply([]any{map[string]any{"text":"I will inspect two locations."},map[string]any{"functionCall":map[string]any{"name":"list_directory","args":map[string]string{"path":"."}}},map[string]any{"functionCall":map[string]any{"name":"list_directory","args":map[string]string{"path":"."}}}}));return}
  _,_=w.Write(reviewReply([]any{map[string]any{"text":"Done."}}))
 }));defer server.Close();a:=reviewAgent(t,server.URL)
 if _,err:=a.Ask(context.Background(),"inspect");err!=nil{t.Fatal(err)}
 calls,results:=[]string{},[]string{}
 for _,e:=range a.Log().Events(){if e.Type=="tool_called"{calls=append(calls,e.Tool.CallID)};if e.Type=="tool_returned"{results=append(results,e.Tool.CallID)}}
 if len(calls)!=2||len(results)!=2||calls[0]==""||calls[0]==calls[1]||calls[0]!=results[0]||calls[1]!=results[1]{t.Fatalf("fallback identity must remain distinct and paired: calls=%v results=%v",calls,results)}
 var dump bytes.Buffer;if err:=a.Log().Dump(&dump);err!=nil{t.Fatal(err)};b:=reviewAgent(t,"");if err:=b.Log().Load(&dump);err!=nil{t.Fatal(err)}
 rendered,err:=b.Engine().Render(b.Log().Context(),b.Engine().Target());if err!=nil{t.Fatal(err)}
 for _,id:=range calls{if bytes.Count(rendered,[]byte(id))!=2{t.Errorf("call/result id %q not replayed as pair",id)}}
}
