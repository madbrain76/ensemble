func makeAgent(t *testing.T) ensemble.Agent {
 t.Helper(); root:=ensemble.New(log.New(io.Discard,"",0)); a,err:=root.NewAgent(ensemble.Config{System:"review",MaxTokens:32});if err!=nil {t.Fatal(err)};return a
}
func TestPublicConstructionAndIsolation(t *testing.T) {
 a:=makeAgent(t);if a.Engine().Parent()!=a || a.Log().Parent()!=a || a.Parent().Agents()[0]!=a {t.Fatal("owner chain")}
 e:=ensemble.Event{Type:"message_received",Message:&ensemble.MessageEvent{Actor:"human",Parts:[]ensemble.Part{{Type:"text",Text:"kept"}}}}
 if err:=a.Log().Append(e);err!=nil {t.Fatal(err)};e.Message.Parts[0].Text="changed"
 c:=a.Log().Context();if c.Dialogue[0].Parts[0].Text!="kept" {t.Fatal("input alias")};c.Dialogue[0].Parts[0].Text="changed"
 b,err:=a.Parent().NewAgent(a.Config());if err!=nil {t.Fatal(err)};if len(b.Log().Events())!=0 || a.Log().Context().Dialogue[0].Parts[0].Text!="kept" {t.Fatal("output/agent isolation")}
}
func TestOpenAIReasoningPartsPreserved(t *testing.T) {
 a:=makeAgent(t);p:=ensemble.Provenance{Vendor:ensemble.OpenAI,Model:"model",Surface:ensemble.ChatCompletions}
 raw:=[]byte(`{"model":"model","choices":[{"message":{"content":"answer","reasoning_content":"opaque-a","reasoning":"opaque-b"}}],"usage":{"prompt_tokens":2,"completion_tokens":3}}`)
 parsed,err:=a.Engine().Parse(raw,p);if err!=nil {t.Fatal(err)}
 if err:=a.Log().Append(ensemble.Event{Type:"response_ended",Response:&parsed});err!=nil {t.Fatal(err)}
 wire,err:=a.Engine().Render(a.Log().Context(),p);if err!=nil {t.Fatal(err)}
 if !bytes.Contains(wire,[]byte(`"reasoning_content":"opaque-a"`)) || !bytes.Contains(wire,[]byte(`"reasoning":"opaque-b"`)) {t.Fatalf("exact-provenance replay lost an opaque field: %s",wire)}
}
func TestGeminiEarlierIDLessResultName(t *testing.T) {
 a:=makeAgent(t);p:=ensemble.Provenance{Vendor:ensemble.Gemini,Model:"model",Surface:ensemble.GenerateContent}
 for _,name:=range []string{"first_tool","second_tool"} {
  resp:=ensemble.ResponseEvent{From:p,Parts:[]ensemble.Part{{Type:"tool_call",Name:name,Args:json.RawMessage(`{}`),From:p}}}
  if err:=a.Log().Append(ensemble.Event{Type:"response_ended",Response:&resp});err!=nil {t.Fatal(err)}
  if err:=a.Log().Append(ensemble.Event{Type:"tool_returned",Tool:&ensemble.ToolEvent{Parts:[]ensemble.Part{{Type:"text",Text:name+" result"}}}});err!=nil {t.Fatal(err)}
 }
 wire,err:=a.Engine().Render(a.Log().Context(),p);if err!=nil {t.Fatal(err)}
 var req struct {Contents []struct {Parts []struct {Result *struct {Name string `json:"name"`} `json:"functionResponse"`} `json:"parts"`} `json:"contents"`}
 if err:=json.Unmarshal(wire,&req);err!=nil {t.Fatal(err)}
 var got []string;for _,c:=range req.Contents {for _,part:=range c.Parts {if part.Result!=nil {got=append(got,part.Result.Name)}}}
 if len(got)!=2 || got[0]!="first_tool" || got[1]!="second_tool" {t.Fatalf("result names %q; earlier result was rebound to later call: %s",got,wire)}
}
func TestMeasuredFailureAccountingAcrossVendors(t *testing.T) {
 for _,tc:=range []struct{vendor ensemble.Vendor;body string}{
  {ensemble.OpenAI,`{"choices":[],"usage":{"prompt_tokens":11,"completion_tokens":2}}`},
  {ensemble.Gemini,`{"candidates":[],"usageMetadata":{"promptTokenCount":11,"candidatesTokenCount":2}}`},
 } {t.Run(tc.vendor.String(),func(t *testing.T){
  server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){io.WriteString(w,tc.body)}));defer server.Close()
  a,err:=ensemble.New(log.New(io.Discard,"",0)).NewAgent(ensemble.Config{Vendor:tc.vendor,BaseURL:server.URL,APIKey:"local-fixture",Model:"model",System:"review",MaxTokens:32});if err!=nil {t.Fatal(err)}
  if _,err:=a.Ask(context.Background(),"question");err==nil {t.Fatal("unusable response accepted")}
  if u:=a.Engine().Usage();u!=(ensemble.Usage{Input:11,Output:2}) {t.Fatalf("measured unusable-response usage lost: got %+v, want input=11 output=2",u)}
  if len(a.History())!=0 {t.Fatal("failed answer committed")}
 })}
}
