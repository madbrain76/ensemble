type reviewSocket struct { c net.Conn; r *bufio.Reader }
func reviewConnect(t *testing.T, url string) *reviewSocket {
 t.Helper(); c,e:=net.Dial("tcp",strings.TrimPrefix(url,"http://")); if e!=nil {t.Fatal(e)}; t.Cleanup(func(){c.Close()})
 c.SetDeadline(time.Now().Add(12*time.Second)); fmt.Fprintf(c,"GET /ws HTTP/1.1\r\nHost: %s\r\nConnection: Upgrade\r\nUpgrade: websocket\r\nSec-WebSocket-Version: 13\r\nSec-WebSocket-Key: AAECAwQFBgcICQoLDA0ODw==\r\n\r\n",strings.TrimPrefix(url,"http://"))
 r:=bufio.NewReader(c); res,e:=http.ReadResponse(r,nil); if e!=nil {t.Fatal(e)}; if res.StatusCode!=101 {t.Fatalf("upgrade %s",res.Status)}; return &reviewSocket{c,r}
}
func (s *reviewSocket) snapshot(t *testing.T) []wire {
 t.Helper(); payload:=[]byte(`{"type":"subscribe"}`); b:=[]byte{0x81,0x80|byte(len(payload)),7,17,23,31}; for i,x:=range payload {b=append(b,x^b[2+i%4])}; if _,e:=s.c.Write(b);e!=nil {t.Fatal(e)}
 var out []wire
 for {var h [2]byte; if _,e:=io.ReadFull(s.r,h[:]);e!=nil {t.Fatal(e)}; n:=uint64(h[1]&127); if n==126 {var x [2]byte;io.ReadFull(s.r,x[:]);n=uint64(binary.BigEndian.Uint16(x[:]))};if n==127 {var x [8]byte;io.ReadFull(s.r,x[:]);n=binary.BigEndian.Uint64(x[:])};if n>4<<20 {t.Fatal("frame too large")};if h[1]&128!=0 {t.Fatal("server masked frame")};p:=make([]byte,n);if _,e:=io.ReadFull(s.r,p);e!=nil {t.Fatal(e)}; if h[0]&15!=1 {continue};var m wire;if e:=json.Unmarshal(p,&m);e!=nil {t.Fatal(e)};out=append(out,m);if m["type"]=="replay_end" {return out}}
}
func reviewCounts(xs []wire) map[string]int {m:=map[string]int{};for _,x:=range xs {typ,_:=x["type"].(string);phase:="live";if x["replay"]==true {phase="replay"};m[phase+":"+typ]++;if typ=="tool_finished"&&x["is_error"]==true {m[phase+":error_result"]++}};return m}
func reviewIDs(xs []wire, typ string) []string {var out []string;for _,x:=range xs {if x["type"]==typ {out=append(out,fmt.Sprint(x["part_id"]))}};return out}
func reviewPairs(a ensemble.Agent) map[string]int {m:=map[string]int{};for _,e:=range a.Log().Events() {if e.Tool==nil {continue};m[e.Type]++;if e.Type=="tool_called"&&e.Tool.Job==nil {m["called_without_job"]++};if e.Type=="tool_returned"&&e.Tool.Result.IsError {m["error_results"]++}};return m}

func TestReviewerR1RealWire(t *testing.T) {
 dir:=t.TempDir(); os.WriteFile(filepath.Join(dir,"seed.txt"),[]byte("READABLE_SEED_CONTENT"),0600)

  {"type":"tool_use","id":"read","name":"read_file","input":map[string]any{"path":"seed.txt"}},
  {"type":"tool_use","id":"write","name":"write_file","input":map[string]any{"path":"written.txt","content":"FILE_EFFECT"}},
  {"type":"tool_use","id":"process","name":"run_command","input":map[string]any{"command":fmt.Sprintf("printf PROCESS_EFFECT > %q",processMark),"ai_callback_delay":0.1}},

  n:=requests.Add(1);var content []map[string]any;switch n {case 1:content=calls;case 2:close(refusedReady);select {case <-retry:case <-r.Context().Done():return};content=append([]map[string]any{{"type":"text","text":"I will run these local checks and observe their results."}},calls...);default:content=[]map[string]any{{"type":"text","text":"Done."}}}
  w.Header().Set("Content-Type","application/json");json.NewEncoder(w).Encode(map[string]any{"id":"local-review","type":"message","role":"assistant","content":content,"stop_reason":"end_turn","usage":map[string]int{"input_tokens":1,"output_tokens":1}})
 }));defer vendor.Close()
 cfg:=ensemble.Config{Vendor:ensemble.VendorAnthropic,Model:"claude-sonnet-4-6",APIKey:"offline-fixture",BaseURL:vendor.URL,DataDir:filepath.Join(dir,"data"),WorkspaceDir:dir,EnableTools:true,DisableStreaming:true,RequireToolExplanation:true,MaxTokens:512}
 root:=ensemble.NewRoot(cfg);defer root.Shutdown(context.Background(),ensemble.ShutdownOptions{});a:=root.NewAgent("review")

 h:=newHub(root,a);server:=httptest.NewServer(http.HandlerFunc(h.websocket));defer server.Close();live:=reviewConnect(t,server.URL);live.snapshot(t)
 ctx,cancel:=context.WithTimeout(context.Background(),20*time.Second);defer cancel();defer allow();done:=make(chan error,1)
 go func(){ans,e:=a.Ask(ctx,"exercise refusal and an explained retry");if e==nil&&ans!="Done." {e=fmt.Errorf("unexpected answer %q",ans)};done<-e}()
 select {case <-refusedReady:case <-ctx.Done():t.Fatal(ctx.Err())}
 first:=live.snapshot(t);reconnect:=reviewConnect(t,server.URL);firstFresh:=reconnect.snapshot(t);firstCounts:=reviewCounts(first);firstFacts:=reviewPairs(a);_,fe:=os.Stat(fileMark);_,pe:=os.Stat(processMark)
 if admissions.Load()!=0||custom.Load()!=0||!os.IsNotExist(fe)||!os.IsNotExist(pe){t.Error("refused batch was admitted or had a real effect")}

}
