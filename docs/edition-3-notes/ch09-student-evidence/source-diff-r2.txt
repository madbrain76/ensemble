+func skillCall(t *testing.T, a ensemble.Agent, name string, args any, wantError bool) string {
+	t.Helper()
+	raw, err := json.Marshal(args)
+	if err != nil {
+		t.Fatal(err)
+	}
+	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
+	defer cancel()
+	text, err := a.CallTool(ctx, name, raw)
+	if err != nil {
+		t.Fatalf("CallTool transport %s: %v", name, err)
+	}
+	facts := a.Log().Events()
+	var result *ensemble.ToolEvent
+	for _, e := range facts {
+		if e.Type == "tool_returned" {
+			result = e.Tool
+		}
+	}
+	if result == nil || result.Name != name || result.IsError != wantError {
+		t.Fatalf("%s result=%+v text=%s", name, result, text)
+	}
+	return text

+package skills

+import (
+	"fmt"

+type definition struct {
+	name, kind, description, body, path string
+	tools, offers, depends              []string
+}
+
+// parse accepts only scalar fields and whitespace/block lists, not general YAML.
+// Boundaries must occupy a line; horizontal rules after frontmatter stay in body.
+func parse(path string, data []byte) (definition, error) {
+	d := definition{path: path}
+	fail := func(message string) (definition, error) { return d, fmt.Errorf("%s: %s", path, message) }
+	text := strings.ReplaceAll(string(data), "\r\n", "\n")
+	lines := strings.SplitAfter(text, "\n")
+	if len(lines) < 3 || strings.TrimSpace(lines[0]) != "---" {
+		return fail("missing opening frontmatter boundary")
+	}
+	end := 1
+	for end < len(lines) && strings.TrimSpace(lines[end]) != "---" {
+		end++
+	}
+	if end == len(lines) {
+		return fail("missing closing frontmatter boundary")
+	}
+	d.body = strings.Join(lines[end+1:], "")
+	seen := map[string]bool{}
+	var list *[]string
+	for _, line := range lines[1:end] {
+		s := strings.TrimSpace(line)
+		if s == "" || strings.HasPrefix(s, "#") {
+			continue
+		}
+		if strings.HasPrefix(s, "- ") {
+			if list == nil || !strings.HasPrefix(line, " ") {
+				return fail("list item without list field")
+			}
+			v := strings.TrimSpace(s[2:])
+			if v == "" || len(strings.Fields(v)) != 1 {
+				return fail("invalid list item")
+			}
+			*list = append(*list, v)
+			continue
+		}
+		key, value, ok := strings.Cut(s, ":")
+		if !ok || strings.TrimSpace(key) != key || seen[key] {
+			return fail("malformed or duplicate field: " + s)
+		}
+		seen[key] = true
+		value = strings.TrimSpace(value)
+		list = nil
+		switch key {
+		case "name":
+			d.name = value
+		case "type":
+			d.kind = value
+		case "description":
+			d.description = value
+		case "tools":
+			list = &d.tools
+		case "depends":
+			list = &d.depends
+		case "loadable-skills":
+			list = &d.offers
+		default:
+			return fail("unknown field: " + key)
+		}
+		if list != nil {
+			*list = strings.Fields(value)
+		}
+	}
+	if d.name == "" || len(strings.Fields(d.name)) != 1 {
+		return fail("name must be one nonempty identifier")
+	}
+	if d.kind != "primary" && d.kind != "loadable" && d.kind != "dependency" {
+		return fail("type must be primary, loadable or dependency")
+	}
+	return d, nil
+}

+func discover(dir string) (map[string]definition, error) {
+	entries, err := os.ReadDir(dir)
+	if err != nil {
+		return nil, fmt.Errorf("skills directory %s: %w", dir, err)
+	}
+	defs := map[string]definition{}
+	for _, entry := range entries {
+		if !entry.IsDir() {
+			continue
+		}
+		path := filepath.Join(dir, entry.Name(), "SKILL.md")
+		data, err := os.ReadFile(path)
+		if os.IsNotExist(err) {
+			continue

+		}
+		d, err := parse(path, data)

+		}

+	}

+	r.mu.RLock()
+	defer r.mu.RUnlock()
+	var names []string
+	for name := range r.loaded {
+		names = append(names, name)
+	}
+	sort.Strings(names)
+	return names
+}
+func (r *Registry) loadable() []string {
+	set := map[string]bool{}
+	for name := range r.loaded {
+		for _, offer := range r.defs[name].offers {
+			if d, ok := r.defs[offer]; ok && d.kind == "loadable" && !r.loaded[offer] {
+				set[offer] = true
+			}
+		}
+	}
+	var names []string
+	for name := range set {
+		names = append(names, name)
+	}
+	sort.Strings(names)
+	return names
+}
+func (r *Registry) LoadableSkills() []string { r.mu.RLock(); defer r.mu.RUnlock(); return r.loadable() }

+var variable = regexp.MustCompile(`\$[A-Za-z_][A-Za-z0-9_]*`)
+var variableName = regexp.MustCompile(`^[A-Za-z_][A-Za-z0-9_]*$`)
+
+func (r *Registry) RegisterVar(name string, fn func() string) error {
+	if !variableName.MatchString(name) || fn == nil || name == "TOOLS" || name == "SKILLS" {
+		return fmt.Errorf("invalid or reserved skill variable %q", name)
+	}
+	r.mu.Lock()
+	r.vars[name] = fn
+	r.mu.Unlock()
+	return nil
+}
+func (r *Registry) render(bodies []string) string {
+	r.mu.RLock()
+	vars := make(map[string]func() string, len(r.vars))
+	for n, f := range r.vars {
+		vars[n] = f
+	}
+	r.mu.RUnlock()
+	// Replacements are single-pass: callback output is not new template input.
+	return variable.ReplaceAllStringFunc(strings.Join(bodies, "\n\n"), func(s string) string {
+		if f := vars[s[1:]]; f != nil {
+			return f()
+		}
+		return s
+	})
+}
+func (r *Registry) resolve(name string) ([]string, error) {
+	visiting, done := map[string]bool{}, map[string]bool{}
+	var order []string
+	var visit func(string) error
+	visit = func(n string) error {
+		d, ok := r.defs[n]
+		if !ok {
+			return fmt.Errorf("skill %q: missing dependency %q", name, n)
+		}
+		if visiting[n] {
+			return fmt.Errorf("%s: dependency cycle at %q", d.path, n)
+		}
+		if done[n] {
+			return nil
+		}
+		visiting[n] = true
+		for _, dep := range d.depends {
+			if child, ok := r.defs[dep]; ok && child.kind == "primary" {
+				return fmt.Errorf("%s: primary %q cannot be a dependency", d.path, dep)
+			}
+			if err := visit(dep); err != nil {
+				return err
+			}
+		}
+		for _, tool := range d.tools {
+			if !r.installed[tool] {
+				return fmt.Errorf("%s: unknown tool %q", d.path, tool)
+			}
+		}
+		visiting[n] = false
+		done[n] = true
+		order = append(order, n)
+		return nil
+	}
+	err := visit(name)
+	return order, err
+}

+// activate is called only after the full graph validates under mu.
+func (r *Registry) activate(order []string, seq uint64) []string {
+	var bodies []string
+	for _, name := range order {
+		if r.loaded[name] {
+			continue
+		}
+		r.loaded[name] = true
+		d := r.defs[name]
+		bodies = append(bodies, d.body)
+		for _, tool := range d.tools {
+			if _, ok := r.enabled[tool]; !ok {
+				r.enabled[tool] = common.ToolSource{Skill: name, Seq: seq}
+			}
+		}
+	}
+	return bodies
+}
+func (r *Registry) LoadInitial(name string) (string, error) {
+	r.mu.Lock()
+	if r.initial {
+		r.mu.Unlock()
+		return "", fmt.Errorf("primary already loaded")
+	}
+	d, ok := r.defs[name]
+	if !ok || d.kind != "primary" {
+		r.mu.Unlock()
+		return "", fmt.Errorf("skill %q: selected primary not found or wrong type", name)
+	}
+	order, err := r.resolve(name)
+	if err != nil {
+		r.mu.Unlock()
+		return "", err
+	}
+	bodies := r.activate(order, 0)
+	for _, tool := range []string{"load_skill", "unload_skill"} {
+		if _, ok := r.enabled[tool]; !ok {
+			r.enabled[tool] = common.ToolSource{Skill: name}
+		}
+	}
+	r.initial = true
+	r.mu.Unlock()
+	return r.render(bodies), nil
+}
+func (r *Registry) LoadDynamic(name string, seq uint64) (string, error) {
+	r.mu.Lock()
+	d, ok := r.defs[name]
+	if !ok || d.kind != "loadable" {
+		r.mu.Unlock()
+		return "", fmt.Errorf("skill %q is not an available loadable skill", name)
+	}
+	if r.loaded[name] {
+		r.mu.Unlock()
+		return "Skill " + name + " is already loaded.", nil
+	}
+	available := false
+	for _, offer := range r.loadable() {
+		if offer == name {
+			available = true
+		}
+	}
+	if !available {
+		r.mu.Unlock()
+		return "", fmt.Errorf("skill %q is not available", name)
+	}
+	order, err := r.resolve(name)
+	if err != nil {
+		r.mu.Unlock()
+		return "", err
+	}
+	bodies := r.activate(order, seq)
+	r.mu.Unlock()
+	return r.render(bodies), nil
+}
+func (r *Registry) Unload(name string) (string, error) {
+	r.mu.Lock()
+	defer r.mu.Unlock()
+	if !r.loaded[name] {
+		return "", fmt.Errorf("skill %q is not loaded", name)
+	}
+	r.pending[name] = true
+	return "Skill " + name + " is pending unload; its instructions and tools remain active until compaction.", nil
+}

+// Install records a newly registered implementation without granting capability.
+func (r *Registry) Install(name string) { r.mu.Lock(); r.installed[name] = true; r.mu.Unlock() }
