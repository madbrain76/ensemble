type lockedBuffer struct { mu sync.Mutex; b bytes.Buffer }
func (b *lockedBuffer) Write(p []byte)(int,error){b.mu.Lock();defer b.mu.Unlock();return b.b.Write(p)}
func (b *lockedBuffer) String()string{b.mu.Lock();defer b.mu.Unlock();return b.b.String()}

func root(t *testing.T) (*ensemble.Ensemble, ensemble.Agent, *lockedBuffer) {
 t.Helper(); logs:=new(lockedBuffer); e:=ensemble.New(log.New(logs,"",0)); d:=t.TempDir()
 a,err:=e.NewAgent(ensemble.Config{System:"public review",MaxTokens:100,Vendor:ensemble.Anthropic,Model:"claude-sonnet-4-6",ResolvedModel:"claude-sonnet-4-6",EnableTools:true,Workspace:d,DataDir:filepath.Join(d,"data")});if err!=nil {t.Fatal(err)}
 t.Cleanup(func(){if err:=e.Shutdown();err!=nil {t.Error(err)}})
 return e,a,logs
}
func call(t *testing.T,a ensemble.Agent,name string,args any) string {
 t.Helper();raw,err:=json.Marshal(args);if err!=nil {t.Fatal(err)};out,err:=a.CallTool(context.Background(),name,raw);if err!=nil {t.Fatalf("%s: %v",name,err)};return out
}
func writeLarge(t *testing.T,a ensemble.Agent) string {
 t.Helper();p:=filepath.Join(a.Config().Workspace,"large");if err:=os.WriteFile(p,[]byte("HEAD\n"+strings.Repeat("x",40000)+"\nTAIL\n"),0600);err!=nil {t.Fatal(err)};return p
}
func TestPublicPendingRearm(t *testing.T) {
 _,a,_:=root(t);p:=writeLarge(t,a)
 call(t,a,"tool_limits",map[string]any{"max_output_bytes":1024})
 second:=call(t,a,"tool_limits",map[string]any{"ai_callback_delay":0.3})
 out:=call(t,a,"read_file",map[string]any{"path":p})
 t.Logf("second=%s; following read bytes=%d",second,len(out))
 if !strings.Contains(second,"consumed by this tool_limits call") {t.Fatal("old setting not reported consumed")}
 if len(out)<15000 {t.Fatalf("old 1024-byte pending cap survived consumption by tool_limits; following read=%d bytes, want default 16KiB payload",len(out))}
}

func TestPublicPendingResetOverrideAndIsolation(t *testing.T) {
 e,a,_:=root(t);p:=writeLarge(t,a);b,err:=e.NewAgent(ensemble.Config{System:"public review",MaxTokens:100,Vendor:ensemble.Anthropic,Model:"claude-sonnet-4-6",EnableTools:true,Workspace:a.Config().Workspace,DataDir:filepath.Join(t.TempDir(),"data")});if err!=nil{t.Fatal(err)}
 call(t,a,"tool_limits",map[string]any{"max_output_bytes":1024});other:=call(t,b,"read_file",map[string]any{"path":p});if len(other)<15000{t.Fatal("pending leaked to other agent")}
 first:=call(t,a,"read_file",map[string]any{"path":p});if len(first)>2000||!strings.Contains(first,"HEAD")||!strings.Contains(first,"TAIL"){t.Fatal("pending/head-tail failure")}
 again:=call(t,a,"read_file",map[string]any{"path":p});if len(again)<15000{t.Fatal("ordinary reset failure")}
 call(t,a,"tool_limits",map[string]any{"max_output_bytes":1024});over:=call(t,a,"run_command",map[string]any{"command":"head -c 30000 /dev/zero | tr '\\0' x","max_output_bytes":6000});if len(over)<5000||len(over)>7000{t.Fatalf("override length %d",len(over))}
 call(t,a,"tool_limits",map[string]any{"max_output_bytes":1024});if s,err:=a.CallTool(context.Background(),"missing",json.RawMessage(`{}`));err!=nil||!strings.Contains(s,"unknown tool"){t.Fatalf("missing receipt %q %v",s,err)};after:=call(t,a,"read_file",map[string]any{"path":p});if len(after)<15000{t.Fatal("failure did not consume pending")}
}
func TestPublicPluginLifecycleCursorAndTerminalHistory(t *testing.T) {
 e,a,logs:=root(t);release:=make(chan struct{});finished:=make(chan struct{})
 if err:=a.RegisterTool(ensemble.ToolDefinition{Name:"held",Description:"uncancellable external handler",Parameters:json.RawMessage(`{"type":"object"}`)},func(c ensemble.ToolContext,_ json.RawMessage)(string,error){c.Agent.Parent().Logger().Print("public-review-child");<-release;close(finished);return "late-content",nil});err!=nil{t.Fatal(err)}
 call(t,a,"tool_limits",map[string]any{"ai_callback_delay":0});out:=call(t,a,"held",map[string]any{});if !strings.Contains(out,"running"){t.Fatal(out)}
 j,getErr:=e.Jobs().Get(1);if getErr!=nil{t.Fatal("registered handler has no job")};deadline:=time.Now().Add(time.Second);for !strings.Contains(logs.String(),"public-review-child")&&time.Now().Before(deadline){time.Sleep(time.Millisecond)}
 waiter:=make(chan string,1);go func(){s,_:=j.Wait(context.Background(),ensemble.WaitOptions{Delay:30*time.Second,MaxBytes:16384});waiter<-s}()
 if err:=j.Kill("kill_job");err!=nil{t.Fatal(err)};select{case s:=<-waiter:if !strings.Contains(s,"killed"){t.Fatal(s)};case <-time.After(time.Second):t.Fatal("waiter stayed blocked")}
 close(release);<-finished;time.Sleep(20*time.Millisecond)
 again,err:=j.Wait(context.Background(),ensemble.WaitOptions{MaxBytes:16384});if err!=nil||!strings.Contains(again,"killed")||strings.Contains(again,"late-content"){t.Fatalf("late state: %q %v",again,err)}
 disk,err:=os.ReadFile(filepath.Join(a.Config().DataDir,j.Snapshot().Output.Locator));if err!=nil||strings.Contains(string(disk),"late-content"){t.Fatal("late output persisted")}
 called,returned,killed:=false,false,false
 for _,ev:=range a.Log().Events(){if ev.Tool!=nil&&ev.Tool.Name=="held"{if ev.Tool.Job==nil{t.Fatal("registered tool lacks metadata")};called=called||ev.Type=="tool_called";returned=returned||ev.Type=="tool_returned"};if ev.Type=="job_killed"&&ev.Job.Handle==1&&ev.Job.Reason=="kill_job"&&ev.Job.Status=="killed"{killed=true}}
 if !called||!returned||!killed||!strings.Contains(logs.String(),"public-review-child"){t.Fatal("plugin ownership/history missing")}
 // Completed output is consumed once; independent root handles/files stay isolated.
 e2,a2,_:=root(t);call(t,a2,"read_file",map[string]any{"path":writeLarge(t,a2)});j2,_:=e2.Jobs().Get(1);s,_:=j2.Wait(context.Background(),ensemble.WaitOptions{MaxBytes:16384});if strings.Contains(s,"HEAD")||strings.Contains(s,"TAIL"){t.Fatal("completed output replay")};if j.Snapshot().Status!="killed"||j2.Snapshot().Status!="done"{t.Fatal("roots shared terminal state")}
}
func ptyKill(t *testing.T,n int) {
 t.Helper();e,a,_:=root(t);out:=call(t,a,"run_command",map[string]any{"command":"printf 'PID=%s\\nREADY\\n' \"$$\"; sleep 30","ai_callback_pattern":"READY","ai_callback_delay":2})
 m:=regexp.MustCompile(`PID=([0-9]+)`).FindStringSubmatch(out);if len(m)!=2{t.Fatalf("no process ID: %q",out)};pid,_:=strconv.Atoi(m[1]);j,_:=e.Jobs().Get(1)
 // This process group was created solely by this test. Rescue only that group.
 inputDone:=make(chan error,1);go func(){inputDone<-j.Input(strings.Repeat("x",n))}();time.Sleep(150*time.Millisecond)
 inputBlocked:=false;select{case err:=<-inputDone:if err!=nil{t.Fatal(err)};default:inputBlocked=true}
 killDone:=make(chan error,1);start:=time.Now();go func(){killDone<-j.Kill("kill_job")}()
 select {
 case err:=<-killDone:if err!=nil{t.Fatal(err)};t.Logf("input bytes=%d blocked=%v kill elapsed=%s",n,inputBlocked,time.Since(start))
 case <-time.After(2*time.Second):
  t.Logf("input bytes=%d blocked=%v; Kill still blocked after 2s; rescuing owned process group %d",n,inputBlocked,pid)
  if err:=syscall.Kill(-pid,syscall.SIGKILL);err!=nil&&err!=syscall.ESRCH{t.Fatal(err)}
  select{case <-killDone:case <-time.After(3*time.Second):t.Fatal("kill failed after rescue")}
  if inputBlocked{select{case <-inputDone:case <-time.After(3*time.Second):t.Fatal("input failed after rescue")}}
  t.Error("supervision cannot stop this owned job during blocked input without external process-group rescue")
 }
 if j.Snapshot().Status!="killed"{t.Fatal("terminal state wrong")}
 if syscall.Kill(pid,0)==nil{t.Fatal("own shell still exists after kill/reap")}
}
func TestPublicPTYNormalInputKillControl(t *testing.T){ptyKill(t,1)}
func TestPublicPTYBlockedInputKill(t *testing.T){ptyKill(t,256*1024)}

 out=call(t,a,"run_command",map[string]any{"command":"printf 'OLDMARK'; sleep 0.5; printf 'NEWMARK'","ai_callback_pattern":"OLDMARK","ai_callback_delay":2});j,_:=e.Jobs().Get(3)
 start:=time.Now();s,err:=j.Wait(context.Background(),ensemble.WaitOptions{Delay:120*time.Millisecond,Pattern:"OLDMARK",MaxBytes:1000});if err!=nil||time.Since(start)<90*time.Millisecond||strings.Contains(s,"OLDMARK"){t.Fatalf("stale pattern: %s %v %s",s,err,time.Since(start))}
