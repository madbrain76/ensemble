// Evidence-only caller of the existing grading API; not a grading implementation.
package main
import (
 "encoding/json"
 "flag"
 "fmt"
 "os"
 "github.com/waywardgeek/ensemble/internal/gradeedition2"
)
func main() {
 bin:=flag.String("bin","","frozen CLI")
 core:=flag.String("core","","source module")
 consumer:=flag.String("consumer","","external consumer")
 workflow:=flag.String("workflow","","actor exercise")
 exercise:=flag.String("exercise","","streaming exercise (may deliberately be absent for nonsubmission baseline)")
 model:=flag.String("model","claude-sonnet-4-6","known streaming model")
 casePath:=flag.String("case","","consumer case")
 mediaPath:=flag.String("media","","media models")
 flag.Parse()
 c,e:=gradeedition2.ReadConsumerCase(*casePath); if e!=nil {fmt.Fprintln(os.Stderr,e);os.Exit(2)}
 m,e:=gradeedition2.ReadMediaModels(*mediaPath); if e!=nil {fmt.Fprintln(os.Stderr,e);os.Exit(2)}
 r,e:=gradeedition2.Chapter6(*bin,*core,*consumer,*workflow,*exercise,*model,c,m)
 if e!=nil {fmt.Fprintln(os.Stderr,e);os.Exit(2)}
 if e=json.NewEncoder(os.Stdout).Encode(r);e!=nil {fmt.Fprintln(os.Stderr,e);os.Exit(2)}
 for _,c:=range r.Checks {if !c.Passed {os.Exit(1)}}
}
