func agent(t *testing.T, v ensemble.Vendor, url string, disabled bool) (*ensemble.Ensemble, ensemble.Agent) {
	t.Helper()
	r := ensemble.New(log.New(io.Discard, "", 0))
	models := map[ensemble.Vendor]string{ensemble.Anthropic: "claude-sonnet-4-6", ensemble.OpenAI: "gpt-4.1-2025-04-14", ensemble.Gemini: "gemini-3.8-flash"}
	a, e := r.NewAgent(ensemble.Config{Vendor: v, Model: models[v], APIKey: "review-key-sentinel", BaseURL: url, System: "review", MaxTokens: 256, DisableStreaming: disabled, DataDir: t.TempDir()})
	if e != nil {
		t.Fatal(e)
	}
	t.Cleanup(func() {
		if e := r.Shutdown(); e != nil {
			t.Error(e)
		}
	})
	return r, a
}
func encode(x any) string {
	b, e := json.Marshal(x)
	if e != nil {
		panic(e)
	}
	return string(b)
}
func frame(x any) string { return "data: " + encode(x) + "\n\n" }
func parse(t *testing.T, a ensemble.Agent, s, ct string) (ensemble.ResponseEvent, []uint64, map[uint64]ensemble.Part, error) {
	t.Helper()
	var response ensemble.ResponseEvent
	var ids []uint64
	finals := map[uint64]ensemble.Part{}
	err := a.Engine().Parse(&http.Response{StatusCode: 200, Header: http.Header{"Content-Type": []string{ct}}, Body: io.NopCloser(strings.NewReader(s))}, ensemble.StreamCallbacks{OnDelta: func(id uint64, _ ensemble.DeltaKind, _ string) { ids = append(ids, id) }, OnPartFinal: func(id uint64, p ensemble.Part) {
		if _, ok := finals[id]; ok {
			t.Errorf("duplicate final id %d", id)
		}
		finals[id] = p
	}, OnEvent: func(e ensemble.Event) {
		if e.Response != nil {
			response = *e.Response
		}
	}})
	return response, ids, finals, err
}
func canon(x any) any {
	var v any
	if e := json.Unmarshal([]byte(encode(x)), &v); e != nil {
		panic(e)
	}
	return v
}
func gbody(parts []any, done bool) map[string]any {
	c := map[string]any{"content": map[string]any{"parts": parts}}
	if done {
		c["finishReason"] = "STOP"
	}
	return map[string]any{"candidates": []any{c}, "usageMetadata": map[string]any{"promptTokenCount": 0, "candidatesTokenCount": 0}}
}
func TestGeminiSignedAndOpaqueBoundaries(t *testing.T) {
	cases := map[string][]any{
 "standalone_opaque_first": {map[string]any{"thoughtSignature":"review-standalone-opaque"},map[string]any{"text":"visible-after-opaque"}},
		"signed_then_unsigned":                {map[string]any{"text": "signed-one", "thoughtSignature": "review-signature-A"}, map[string]any{"text": "unsigned-two"}},
		"two_signed_text_parts":               {map[string]any{"text": "signed-one", "thoughtSignature": "review-signature-A"}, map[string]any{"text": "signed-two", "thoughtSignature": "review-signature-B"}},
		"signed_thought_then_opaque_boundary": {map[string]any{"text": "review-thought-one", "thought": true, "thoughtSignature": "review-signature-A"}, map[string]any{"thoughtSignature": "review-opaque-boundary"}, map[string]any{"text": "visible-done"}},
	}
	for name, parts := range cases {
		t.Run(name, func(t *testing.T) {
			_, a := agent(t, ensemble.Gemini, "http://unused.invalid", false)
			plain, _, _, e := parse(t, a, encode(gbody(parts, true)), "application/json")
			if e != nil {
				t.Fatal(e)
			}
			var stream string
			for i, p := range parts {
				stream += frame(gbody([]any{p}, i == len(parts)-1))
			}
			got, _, _, e := parse(t, a, stream, "text/event-stream")
			if e != nil {
				t.Fatal(e)
			}
			if !reflect.DeepEqual(canon(got.Parts), canon(plain.Parts)) {
				t.Errorf("canonical parity lost: plain=%s stream=%s", encode(plain.Parts), encode(got.Parts))
			}
			ctx := ensemble.Context{Dialogue: []ensemble.Turn{{Actor: "agent", Parts: got.Parts}}}
			wire, e := a.Engine().Render(&ctx)
			if e != nil {
				t.Fatal(e)
			}
			t.Logf("same-source replay=%s", wire)
		})
	}
}
func TestOpenAISparseInterleavedToolsAndUsage(t *testing.T) {
	_, a := agent(t, ensemble.OpenAI, "http://unused.invalid", false)
	s := frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"content": "I will call two tools."}}}})
	for _, v := range []struct {
		i              int
		id, name, args string
	}{{9, "right", "second", "{\"b\":"}, {2, "left", "first", "{\"a\":"}, {9, "", "", "2}"}, {2, "", "", "1}"}} {
		s += frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"tool_calls": []any{map[string]any{"index": v.i, "id": v.id, "function": map[string]any{"name": v.name, "arguments": v.args}}}}}}})
	}
	s += frame(map[string]any{"choices": []any{}, "usage": map[string]any{"prompt_tokens": 10, "completion_tokens": 0, "prompt_tokens_details": map[string]any{"cached_tokens": 4}}}) + "data: [DONE]"
	r, ids, finals, e := parse(t, a, s, "text/event-stream")
	if e != nil {
		t.Fatal(e)
	}
	if len(r.Parts) != 3 || r.Parts[1].Name != "first" || r.Parts[2].Name != "second" || string(r.Parts[1].Args) != "{\"a\":1}" || string(r.Parts[2].Args) != "{\"b\":2}" {
		t.Fatalf("sparse assembly %+v", r)
	}
	if len(ids) == 0 || ids[0] != 0 || len(finals) != 3 {
		t.Fatalf("zero/correlation ids=%v finals=%v", ids, finals)
	}
	for _, id := range ids {
		if _, ok := finals[id]; !ok {
			t.Fatalf("missing exact final %d", id)
		}
	}
	if r.Usage.Input != 6 || r.Usage.CacheRead != 4 || r.Usage.Output != 0 {
		t.Fatalf("usage %+v", r.Usage)
	}
	_, again, _, e := parse(t, a, `{"choices":[{"message":{"role":"assistant","content":"again"}}],"usage":{"prompt_tokens":0,"completion_tokens":0}}`, "application/json")
	if e != nil || len(again) != 1 || again[0] != 0 {
		t.Fatalf("response-local reset %v %v", again, e)
	}
}
func TestAnthropicIndexedOpaqueAndCacheParity(t *testing.T) {
	_, a := agent(t, ensemble.Anthropic, "http://unused.invalid", false)
	parts := []any{map[string]any{"type": "redacted_thinking", "data": "review-opaque-sentinel"}, map[string]any{"type": "text", "text": "visible"}}
	usage := map[string]any{"input_tokens": 0, "output_tokens": 0, "cache_creation_input_tokens": 3, "cache_read_input_tokens": 7}
	plain := map[string]any{"model": "claude-sonnet-4-6", "content": parts, "usage": usage}
	s := frame(map[string]any{"type": "message_start", "message": map[string]any{"model": "claude-sonnet-4-6", "usage": usage}})
	for i, p := range parts {
		s += frame(map[string]any{"type": "content_block_start", "index": 10 + i*7, "content_block": p}) + frame(map[string]any{"type": "content_block_stop", "index": 10 + i*7})
	}
	s += frame(map[string]any{"type": "message_stop"})
	p, _, _, e := parse(t, a, encode(plain), "application/json")
	if e != nil {
		t.Fatal(e)
	}
	r, ids, f, e := parse(t, a, s, "text/event-stream")
	if e != nil || !reflect.DeepEqual(canon(p), canon(r)) {
		t.Fatalf("parity err=%v plain=%s stream=%s", e, encode(p), encode(r))
	}
	if len(f) != 2 || len(ids) != 1 || ids[0] != 17 {
		t.Fatalf("opaque must not become visible delta: %v %v", ids, f)
	}
	if r.Usage.Input != 0 || r.Usage.CacheWrite != 3 || r.Usage.CacheRead != 7 {
		t.Fatal(r.Usage)
	}
}
func TestSSEFramingAndUnknownFields(t *testing.T) {
	_, a := agent(t, ensemble.OpenAI, "http://unused.invalid", false)
	s := ": keepalive\r\nid: 7\r\nretry: 10\r\nignored: x\r\nevent: one\r\ndata: {\"choices\":\r\ndata: [{\"delta\":{\"content\":\"framed\"}}]}\r\n\r\ndata: {\"choices\":[],\"usage\":{\"prompt_tokens\":0,\"completion_tokens\":0}}\r\n\r\ndata: [DONE]"
	var events []string
	var parts []ensemble.Part
	e := a.Engine().Parse(&http.Response{StatusCode: 200, Header: http.Header{"Content-Type": []string{"text/event-stream; charset=utf-8"}}, Body: io.NopCloser(strings.NewReader(s))}, ensemble.StreamCallbacks{OnFrame: func(event string, _ []byte) { events = append(events, event) }, OnPartFinal: func(_ uint64, p ensemble.Part) { parts = append(parts, p) }})
	if e != nil || len(parts) != 1 || parts[0].Text != "framed" || !reflect.DeepEqual(events, []string{"one", "", ""}) {
		t.Fatalf("framing e=%v parts=%+v events=%v", e, parts, events)
	}
}
func TestPartialResponseErrorDoesNotPersistOrAuthorize(t *testing.T) {
	var calls int
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		calls++
		w.Header().Set("Content-Type", "text/event-stream")
		if calls == 1 {
			fmt.Fprint(w, frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"content": "partial-only-review"}}}}))
			return
		}
		fmt.Fprint(w, frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"content": "recovered"}}}, "usage": map[string]any{"prompt_tokens": 0, "completion_tokens": 0}})+"data: [DONE]\n\n")
	}))
	defer srv.Close()
	_, a := agent(t, ensemble.OpenAI, srv.URL, false)
	if _, e := a.Ask(context.Background(), "fail"); e == nil {
		t.Fatal("truncated accepted")
	}
	if x, e := a.Ask(context.Background(), "recover"); e != nil || x != "recovered" {
		t.Fatal(x, e)
	}
	var dump bytes.Buffer
	if e := a.Log().Dump(&dump); e != nil {
		t.Fatal(e)
	}
	if strings.Contains(dump.String(), "partial-only-review") {
		t.Fatal("partial response persisted")
	}
	ends := 0
	for _, e := range a.Log().Events() {
		if e.Type == "response_ended" {
			ends++
		}
	}
	if ends != 1 {
		t.Fatalf("durable response count %d", ends)
	}
}

func watch(a ensemble.Agent) (string, int, int, error) {
	d, f := 0, 0
	text, e := a.AskWatching(context.Background(), "review request", func(o ensemble.Observation) {
		switch x := o.(type) {
		case ensemble.PartDelta:
			if x.Origin == "provider" {
				d++
			}
		case ensemble.PartFinal:
			if x.Response != 0 {
				f++
			}
		}
	})
	return text, d, f, e
}
func TestRealRequestModesAndUnknownFallback(t *testing.T) {
	for _, v := range []ensemble.Vendor{ensemble.Anthropic, ensemble.OpenAI, ensemble.Gemini} {
		for _, disabled := range []bool{false, true} {
			t.Run(v.String()+fmt.Sprint(disabled), func(t *testing.T) {
				var observed bool
				srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
					var req map[string]any
					if e := json.NewDecoder(r.Body).Decode(&req); e != nil {
						t.Error(e)
					}
					observed = true
					if v == ensemble.Gemini {
						if strings.Contains(r.URL.Path, "streamGenerateContent") == disabled || ((r.URL.Query().Get("alt") == "sse") == disabled) {
							t.Errorf("wrong Gemini request mode disabled=%v", disabled)
						}
					} else {
						if (req["stream"] == true) == disabled {
							t.Errorf("wrong request stream flag disabled=%v stream=%v", disabled, req["stream"])
						}
						if v == ensemble.OpenAI && !disabled {
							opts, _ := req["stream_options"].(map[string]any)
							if opts["include_usage"] != true {
								t.Error("missing usage request")
							}
						}
					}
					w.Header().Set("Content-Type", "application/json")
					switch v {
					case ensemble.Anthropic:
						fmt.Fprint(w, `{"content":[{"type":"text","text":"ok"}],"usage":{"input_tokens":0,"output_tokens":0}}`)
					case ensemble.OpenAI:
						fmt.Fprint(w, `{"choices":[{"message":{"role":"assistant","content":"ok"}}],"usage":{"prompt_tokens":0,"completion_tokens":0}}`)
					case ensemble.Gemini:
						fmt.Fprint(w, `{"candidates":[{"content":{"parts":[{"text":"ok"}]}}],"usageMetadata":{"promptTokenCount":0,"candidatesTokenCount":0}}`)
					}
				}))
				defer srv.Close()
				_, a := agent(t, v, srv.URL, disabled)
				text, d, f, e := watch(a)
				if e != nil || text != "ok" || d != 1 || f != 1 || !observed {
					t.Fatalf("text=%q err=%v deltas=%d finals=%d observed=%v", text, e, d, f, observed)
				}
			})
		}
	}
	t.Run("unlisted_model", func(t *testing.T) {
		srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			var req map[string]any
			_ = json.NewDecoder(r.Body).Decode(&req)
			if req["stream"] == true {
				t.Error("unknown model requested streaming")
			}
			w.Header().Set("Content-Type", "application/json")
			fmt.Fprint(w, `{"content":[{"type":"text","text":"unknown-model-ok"}],"usage":{"input_tokens":0,"output_tokens":0}}`)
		}))
		defer srv.Close()
		r := ensemble.New(log.New(io.Discard, "", 0))
		defer r.Shutdown()
		a, e := r.NewAgent(ensemble.Config{Vendor: ensemble.Anthropic, Model: "review-model-not-in-table", System: "review", APIKey: "fixture", BaseURL: srv.URL, MaxTokens: 64})
		if e != nil {
			t.Fatal(e)
		}
		text, d, f, e := watch(a)
		if e != nil || text != "unknown-model-ok" || d != 1 || f != 1 {
			t.Fatalf("unknown text=%q err=%v delta=%d final=%d", text, e, d, f)
		}
	})
}

func TestLateFramesAfterInterruptAreFenced(t *testing.T) {
	early, release, handlerDone := make(chan struct{}), make(chan struct{}), make(chan struct{})
	var requests, late atomic.Int32
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/event-stream")
		if requests.Add(1) == 1 {
			fmt.Fprint(w, frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"content": "early-old-review"}}}}))
			w.(http.Flusher).Flush()
			<-release
			defer close(handlerDone)
			fmt.Fprint(w, frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"content": "late-old-review"}}}, "usage": map[string]any{"prompt_tokens": 0, "completion_tokens": 0}})+"data: [DONE]\n\n")
			return
		}
		fmt.Fprint(w, frame(map[string]any{"choices": []any{map[string]any{"delta": map[string]any{"content": "fresh-review"}}}, "usage": map[string]any{"prompt_tokens": 0, "completion_tokens": 0}})+"data: [DONE]\n\n")
	}))
	defer srv.Close()
	_, a := agent(t, ensemble.OpenAI, srv.URL, false)
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	done := make(chan error, 1)
	go func() {
		_, e := a.AskWatching(ctx, "old", func(o ensemble.Observation) {
			if d, ok := o.(ensemble.PartDelta); ok && d.Origin == "provider" {
				if d.Chunk == "early-old-review" {
					close(early)
				}
				if strings.Contains(d.Chunk, "late-old-review") {
					late.Add(1)
				}
			}
		})
		done <- e
	}()
	select {
	case <-early:
	case <-ctx.Done():
		close(release)
		t.Fatal("no early callback")
	}
	if e := a.Post(ensemble.Interrupt{}); e != nil {
		close(release)
		t.Fatal(e)
	}
	if e := <-done; !errors.Is(e, ensemble.ErrInterrupted) {
		close(release)
		t.Fatalf("old completion %v", e)
	}
	text, e := a.Ask(ctx, "fresh")
	close(release)
	<-handlerDone
	if e != nil || text != "fresh-review" {
		t.Fatal(text, e)
	}
	if late.Load() != 0 {
		t.Fatal("late old delta published")
	}
	var dump bytes.Buffer
	_ = a.Log().Dump(&dump)
	if strings.Contains(dump.String(), "old-review") {
		t.Fatal("interrupted partial response persisted")
	}
	responses := 0
	for _, e := range a.Log().Events() {
		if e.Type == "response_ended" {
			responses++
		}
	}
	if responses != 1 {
		t.Fatalf("durable responses=%d", responses)
	}
}
