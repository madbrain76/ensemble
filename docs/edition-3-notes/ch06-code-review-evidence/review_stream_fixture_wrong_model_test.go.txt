package ensemble_test

import (
	"bytes"
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/http/httptest"
	"testing"
)

// Reviewer fixture adapter only: original actor tests and production code unchanged.
func reviewStreamingServer(t *testing.T, h http.HandlerFunc) *httptest.Server {
	return httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		b, e := io.ReadAll(r.Body)
		if e != nil {
			t.Error(e)
		}
		r.Body = io.NopCloser(bytes.NewReader(b))
		var req map[string]any
		_ = json.Unmarshal(b, &req)
		if req["stream"] != true {
			t.Error("streaming wakeup request lacks stream=true")
		}
		rec := httptest.NewRecorder()
		h(rec, r)
		if rec.Code != 200 {
			w.WriteHeader(rec.Code)
			_, _ = w.Write(rec.Body.Bytes())
			return
		}
		var response map[string]any
		if e := json.Unmarshal(rec.Body.Bytes(), &response); e != nil {
			t.Error(e)
			w.WriteHeader(500)
			return
		}
		parts, ok := response["content"].([]any)
		if !ok {
			t.Errorf("unexpected fixture response schema")
			w.WriteHeader(500)
			return
		}
		w.Header().Set("Content-Type", "text/event-stream")
		emit := func(v any) {
			b, _ := json.Marshal(v)
			_, _ = fmt.Fprintf(w, "data: %s\n\n", b)
			w.(http.Flusher).Flush()
		}
		emit(map[string]any{"type": "message_start", "message": map[string]any{"model": "claude-sonnet-4-6", "usage": response["usage"]}})
		for i, p := range parts {
			part := p.(map[string]any)
			kind, _ := part["type"].(string)
			text, _ := part["text"].(string)
			args := part["input"]
			if kind == "text" {
				part["text"] = ""
			}
			if kind == "tool_use" {
				part["input"] = map[string]any{}
			}
			emit(map[string]any{"type": "content_block_start", "index": i, "content_block": part})
			if kind == "text" {
				mid := len(text) / 2
				for _, chunk := range []string{text[:mid], text[mid:]} {
					emit(map[string]any{"type": "content_block_delta", "index": i, "delta": map[string]any{"type": "text_delta", "text": chunk}})
				}
			}
			if kind == "tool_use" {
				b, _ := json.Marshal(args)
				mid := len(b) / 2
				for _, chunk := range []string{string(b[:mid]), string(b[mid:])} {
					emit(map[string]any{"type": "content_block_delta", "index": i, "delta": map[string]any{"type": "input_json_delta", "partial_json": chunk}})
				}
			}
			emit(map[string]any{"type": "content_block_stop", "index": i})
		}
		emit(map[string]any{"type": "message_delta", "usage": response["usage"]})
		emit(map[string]any{"type": "message_stop"})
		t.Logf("review fixture delivered %d streamed blocks", len(parts))
	}))
}
