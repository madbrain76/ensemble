package reviewchecks

import (
	"context"
	"encoding/json"
	"ensemble"
	"errors"
	"fmt"
	"io"
	"log"
	"net/http"
	"net/http/httptest"
	"sync"
	"sync/atomic"
	"testing"
	"time"
)

func TestStreamingMultiCallInterruptPreservesPairs(t *testing.T) {
	started, release := make(chan struct{}), make(chan struct{})
	var once sync.Once
	closeRelease := func() { once.Do(func() { close(release) }) }
	var calls, effects atomic.Int32
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Header().Set("Content-Type", "text/event-stream")
		emit := func(v any) { fmt.Fprint(w, frame(v)); w.(http.Flusher).Flush() }
		emit(map[string]any{"type": "message_start", "message": map[string]any{"model": "claude-sonnet-4-6", "usage": map[string]any{"input_tokens": 1, "output_tokens": 1}}})
		parts := []any{map[string]any{"type": "text", "text": "fresh-batch-review"}}
		if calls.Add(1) == 1 {
			parts = []any{map[string]any{"type": "text", "text": "I will wait for the first tool then perform the next one."}, map[string]any{"type": "tool_use", "id": "held-review", "name": "held", "input": map[string]any{}}, map[string]any{"type": "tool_use", "id": "later-review", "name": "later", "input": map[string]any{}}}
		}
		for i, p := range parts {
			emit(map[string]any{"type": "content_block_start", "index": i, "content_block": p})
			emit(map[string]any{"type": "content_block_stop", "index": i})
		}
		emit(map[string]any{"type": "message_stop"})
	}))
	defer server.Close()
	root := ensemble.New(log.New(io.Discard, "", 0))
	defer root.Shutdown()
	defer closeRelease()
	a, e := root.NewAgent(ensemble.Config{Vendor: ensemble.Anthropic, Model: "claude-sonnet-4-6", APIKey: "fixture", BaseURL: server.URL, System: "review", MaxTokens: 256, EnableTools: true, RequireToolExplanation: true, ToolBatchLimit: 2, DataDir: t.TempDir()})
	if e != nil {
		t.Fatal(e)
	}
	params := json.RawMessage(`{"type":"object"}`)
	a.RegisterTool(ensemble.ToolDefinition{Name: "held", Description: "wait", Parameters: params}, func(ensemble.ToolContext, json.RawMessage) (string, error) {
		close(started)
		<-release
		return "late-retained-job", nil
	})
	a.RegisterTool(ensemble.ToolDefinition{Name: "later", Description: "side effect", Parameters: params}, func(ensemble.ToolContext, json.RawMessage) (string, error) { effects.Add(1); return "unexpected", nil })
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	reply := a.Submit(ctx, "old batch")
	select {
	case <-started:
	case <-ctx.Done():
		t.Fatal("first tool not started")
	}
	if e := a.Post(ensemble.Interrupt{}); e != nil {
		t.Fatal(e)
	}
	select {
	case result := <-reply:
		if !errors.Is(result.Err, ensemble.ErrInterrupted) {
			t.Fatal(result)
		}
	case <-ctx.Done():
		t.Fatal("interrupt not completed")
	}
	fresh, e := a.Ask(ctx, "new request while retained tool runs")
	if e != nil || fresh != "fresh-batch-review" {
		t.Fatal(fresh, e)
	}
	if effects.Load() != 0 {
		t.Fatal("unstarted batch tool executed")
	}
	job, ok := root.Jobs().Get(1)
	if !ok || job.Snapshot().Status != "running" {
		t.Fatal("interruption killed or lost retained job")
	}
	paired := map[string]int{}
	original := 0
	for _, ev := range a.Log().Events() {
		if ev.Response != nil {
			for _, p := range ev.Response.Parts {
				if p.Type == "tool_call" {
					original++
				}
			}
		}
		if ev.Type == "tool_returned" && ev.Tool != nil {
			paired[ev.Tool.CallID]++
			if ev.Tool.CallID == "later-review" && !ev.Tool.IsError {
				t.Fatal("unstarted tool fabricated success")
			}
		}
	}
	if original != 2 || paired["held-review"] != 1 || paired["later-review"] != 1 {
		t.Fatalf("original=%d paired=%v", original, paired)
	}
	closeRelease()
}
