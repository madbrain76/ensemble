// These are synthetic boundary markers, not provider-issued opaque material.
const boundaryModel = "r1-returned-model"

var boundaryFrom = ensemble.Provenance{Vendor: ensemble.Gemini, Surface: ensemble.GenerateContent, Model: boundaryModel}

type boundaryCase struct {
	name   string
	parts  []string
	frames [][]string
	want   []ensemble.Part
	answer string
}

func boundaryText(text, sig string) ensemble.Part {
	p := ensemble.Part{Type: "text", Text: text}
	if sig != "" {
		p.From = boundaryFrom
		p.Opaque = json.RawMessage(fmt.Sprintf("%q", sig))
	}
	return p
}
func boundaryOpaque(raw string) ensemble.Part {
	return ensemble.Part{Type: "opaque", From: boundaryFrom, Data: json.RawMessage(raw)}
}
func boundaryCases() []boundaryCase {
	return []boundaryCase{
		{name: "opaque-first", parts: []string{`{"thoughtSignature":"r1-opaque-first"}`, `{"text":"visible"}`}, want: []ensemble.Part{boundaryOpaque(`{"thoughtSignature":"r1-opaque-first"}`), boundaryText("visible", "")}, answer: "visible"},
		{name: "signed-then-unsigned", parts: []string{`{"text":"signed ","thoughtSignature":"r1-signed"}`, `{"text":"unsigned"}`}, want: []ensemble.Part{boundaryText("signed ", "r1-signed"), boundaryText("unsigned", "")}, answer: "signed unsigned"},
		{name: "two-signed", parts: []string{`{"text":"first ","thoughtSignature":"r1-first"}`, `{"text":"second","thoughtSignature":"r1-second"}`}, want: []ensemble.Part{boundaryText("first ", "r1-first"), boundaryText("second", "r1-second")}, answer: "first second"},
		{name: "thought-opaque-visible", parts: []string{`{"text":"synthetic-hidden","thought":true,"thoughtSignature":"r1-thought"}`, `{"thoughtSignature":"r1-boundary"}`, `{"text":"visible"}`}, want: []ensemble.Part{boundaryOpaque(`{"text":"synthetic-hidden","thought":true,"thoughtSignature":"r1-thought"}`), boundaryOpaque(`{"thoughtSignature":"r1-boundary"}`), boundaryText("visible", "")}, answer: "visible"},
	}
}
func boundaryContinuations() []boundaryCase {
	return []boundaryCase{
		{name: "text-tail-then-new-text", parts: []string{`{"text":"portable","thoughtSignature":"r1-tail"}`, `{"text":" next"}`}, frames: [][]string{{`{"text":"por"}`}, {`{"text":"table"}`}, {`{"thoughtSignature":"r1-tail"}`}, {`{"text":" next"}`}}, want: []ensemble.Part{boundaryText("portable", "r1-tail"), boundaryText(" next", "")}, answer: "portable next"},
		{name: "thought-tail", parts: []string{`{"text":"synthetic-hidden","thought":true,"thoughtSignature":"r1-tail-thought"}`, `{"text":"visible"}`}, frames: [][]string{{`{"text":"synthetic-","thought":true}`}, {`{"text":"hidden","thought":true}`}, {`{"thoughtSignature":"r1-tail-thought"}`}, {`{"text":"visible"}`}}, want: []ensemble.Part{boundaryOpaque(`{"text":"synthetic-hidden","thought":true,"thoughtSignature":"r1-tail-thought"}`), boundaryText("visible", "")}, answer: "visible"},
		{name: "tool-tail", parts: []string{`{"text":"explanation"}`, `{"functionCall":{"id":"r1-call","name":"lookup","args":{"q":"x"}},"thoughtSignature":"r1-tail-tool"}`}, frames: [][]string{{`{"text":"explanation"}`, `{"functionCall":{"id":"r1-call","name":"lookup","args":{"q":"x"}}}`}, {`{"thoughtSignature":"r1-tail-tool"}`}}, want: []ensemble.Part{boundaryText("explanation", ""), {Type: "tool_call", CallID: "r1-call", Name: "lookup", Args: json.RawMessage(`{"q":"x"}`), From: boundaryFrom, Opaque: json.RawMessage(`"r1-tail-tool"`)}}, answer: "explanation"},
		// An explicit second array element is not a trailing continuation frame.
		{name: "explicit-text-and-opaque", parts: []string{`{"text":"visible"}`, `{"thoughtSignature":"r1-separate"}`}, frames: [][]string{{`{"text":"visible"}`, `{"thoughtSignature":"r1-separate"}`}}, want: []ensemble.Part{boundaryText("visible", ""), boundaryOpaque(`{"thoughtSignature":"r1-separate"}`)}, answer: "visible"},
		{name: "explicit-two-unsigned", parts: []string{`{"text":"first "}`, `{"text":"second"}`}, frames: [][]string{{`{"text":"first "}`, `{"text":"second"}`}}, want: []ensemble.Part{boundaryText("first ", ""), boundaryText("second", "")}, answer: "first second"},
	}
}
func boundaryWire(parts []string, end bool) string {
	tail := ""
	if end {
		tail = `,"finishReason":"STOP"`
	}
	return `{"modelVersion":"` + boundaryModel + `","candidates":[{"content":{"parts":[` + strings.Join(parts, ",") + `]}` + tail + `}],"usageMetadata":{"promptTokenCount":3,"cachedContentTokenCount":1,"candidatesTokenCount":2,"thoughtsTokenCount":1}}`
}
func boundaryServer(t *testing.T, c boundaryCase) *httptest.Server {
	t.Helper()
	s := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		if r.URL.Path == "/plain" || !strings.Contains(r.URL.Path, "stream") {
			w.Header().Set("Content-Type", "application/json")
			fmt.Fprint(w, boundaryWire(c.parts, true))
			return
		}
		w.Header().Set("Content-Type", "text/event-stream")
		for i, p := range c.frames {
			fmt.Fprint(w, frame(boundaryWire(p, i == len(c.frames)-1)))
			w.(http.Flusher).Flush()
		}
	}))
	t.Cleanup(s.Close)
	return s
}
func boundaryAgent(t *testing.T, root *ensemble.Ensemble, vendor ensemble.Vendor, model, url string, disable bool) ensemble.Agent {
	t.Helper()
	a, err := root.NewAgent(ensemble.Config{Vendor: vendor, Model: model, ResolvedModel: model, BaseURL: url, APIKey: "r1-local-fixture-only", System: "test", MaxTokens: 512, DisableStreaming: disable, HistoryPath: filepath.Join(t.TempDir(), "history.jsonl")})
	if err != nil {
		t.Fatal(err)
	}
	return a
}
func boundaryCheck(t *testing.T, a ensemble.Agent, url string, c boundaryCase) ensemble.ResponseEvent {
	t.Helper()
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		t.Fatal(err)
	}
	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		t.Fatal(err)
	}
	defer resp.Body.Close()
	var result ensemble.ResponseEvent
	events := 0
	ids := []uint64{}
	finals := []ensemble.Part{}
	deltas := map[uint64]map[ensemble.DeltaKind]string{}
	err = a.Engine().Parse(resp, ensemble.StreamCallbacks{
		OnEvent: func(e ensemble.Event) {
			if e.Response != nil {
				events++
				result = *e.Response
			}
		},
		OnPartFinal: func(id uint64, p ensemble.Part) { ids = append(ids, id); finals = append(finals, p) },
		OnDelta: func(id uint64, k ensemble.DeltaKind, s string) {
			if deltas[id] == nil {
				deltas[id] = map[ensemble.DeltaKind]string{}
			}
			deltas[id][k] += s
		},
	})
	if err != nil {
		t.Fatalf("public Parse: %v", err)
	}
	want := ensemble.ResponseEvent{From: boundaryFrom, Parts: c.want, Usage: ensemble.Usage{Input: 2, CacheRead: 1, Output: 3}}
	if events != 1 || !reflect.DeepEqual(normal6(result), normal6(want)) {
		t.Fatalf("canonical boundary mismatch: events=%d\ngot=%s\nwant=%s", events, must6(result), must6(want))
	}
	if !reflect.DeepEqual(normal6(finals), normal6(c.want)) {
		t.Fatalf("finals mismatch: %s", must6(finals))
	}
	if len(ids) != len(c.want) {
		t.Fatalf("final IDs=%v", ids)
	}
	for i, p := range c.want {
		if ids[i] != uint64(i) {
			t.Fatalf("response-local IDs must reset including zero: %v", ids)
		}
		wantDelta := map[ensemble.DeltaKind]string{}
		switch p.Type {
		case "text":
			wantDelta[ensemble.DeltaText] = p.Text
		case "tool_call":
			wantDelta[ensemble.DeltaToolCall] = p.Name + string(p.Args)
		case "opaque":
			var raw map[string]any
			_ = json.Unmarshal(p.Data, &raw)
			if raw["thought"] == true {
				wantDelta[ensemble.DeltaThinking], _ = raw["text"].(string)
			}
		}
		got := deltas[uint64(i)]
		if got == nil {
			got = map[ensemble.DeltaKind]string{}
		}
		if !reflect.DeepEqual(got, wantDelta) {
			t.Fatalf("part %d delta correlation: got=%v want=%v", i, got, wantDelta)
		}
	}
	for id := range deltas {
		if id >= uint64(len(c.want)) {
			t.Fatalf("unfinalized delta ID %d", id)
		}
	}
	return result
}
func boundaryReplay(t *testing.T, root *ensemble.Ensemble, a ensemble.Agent, result ensemble.ResponseEvent, c boundaryCase) {
	t.Helper()
	if err := a.Log().Append(ensemble.Event{Type: "response_ended", Response: &result}); err != nil {
		t.Fatal(err)
	}
	ctx := a.Log().Context()
	wire, err := a.Engine().Render(&ctx)
	if err != nil {
		t.Fatal(err)
	}
	var request struct {
		Contents []struct {
			Role  string            `json:"role"`
			Parts []json.RawMessage `json:"parts"`
		} `json:"contents"`
	}
	if err = json.Unmarshal(wire, &request); err != nil {
		t.Fatal(err)
	}
	var replay []json.RawMessage
	for _, m := range request.Contents {
		if m.Role == "model" {
			replay = append(replay, m.Parts...)
		}
	}
	wantRaw := []json.RawMessage{}
	for _, p := range c.parts {
		wantRaw = append(wantRaw, json.RawMessage(p))
	}
	if !reflect.DeepEqual(normal6(replay), normal6(wantRaw)) {
		t.Fatalf("same-source binding/order: %s want %s", must6(replay), must6(wantRaw))
	}
	for _, target := range []struct {
		name          string
		vendor        ensemble.Vendor
		model         string
		changeSurface bool
	}{
		{"model", ensemble.Gemini, "r1-other-model", false},
		{"vendor", ensemble.Anthropic, boundaryModel, false},
		{"surface", ensemble.Gemini, boundaryModel, true},
	} {
		t.Run("drop-"+target.name, func(t *testing.T) {
			other := boundaryAgent(t, root, target.vendor, target.model, "http://unused.invalid", true)
			contextCopy := a.Log().Context()
			if target.changeSurface {
				for i := range contextCopy.Dialogue {
					for j := range contextCopy.Dialogue[i].Parts {
						p := &contextCopy.Dialogue[i].Parts[j]
						if p.From.Vendor == ensemble.Gemini {
							p.From.Surface = ensemble.Messages
						}
					}
				}
			}
			out, err := other.Engine().Render(&contextCopy)
			if err != nil {
				t.Fatal(err)
			}
			if strings.Contains(string(out), "r1-") || strings.Contains(string(out), "synthetic-hidden") {
				// Request model names are expected to include r1; inspect only replay payload.
				var request map[string]json.RawMessage
				_ = json.Unmarshal(out, &request)
				payload := string(request["contents"]) + string(request["messages"])
				for _, marker := range []string{"r1-opaque", "r1-signed", "r1-first", "r1-second", "r1-thought", "r1-boundary", "r1-tail", "r1-separate", "synthetic-hidden"} {
					if strings.Contains(payload, marker) {
						t.Fatalf("cross-source private material %q in %s", marker, payload)
					}
				}
			}
			for _, p := range c.want {
				if p.Type == "text" && !strings.Contains(string(out), p.Text) {
					t.Fatalf("ordinary text lost across %s: %s", target.name, out)
				}
			}
		})
	}
}
func TestChapter6GeminiBoundaryParity(t *testing.T) {
	for _, c := range boundaryCases() {
		for _, layout := range []string{"one-frame", "split-frames"} {
			c := c
			t.Run(c.name+"/"+layout, func(t *testing.T) {
				if layout == "one-frame" {
					c.frames = [][]string{c.parts}
				} else {
					for _, p := range c.parts {
						c.frames = append(c.frames, []string{p})
					}
				}
				boundaryPair(t, c)
			})
		}
	}
}
func TestChapter6GeminiBoundaryContinuation(t *testing.T) {
	for _, c := range boundaryContinuations() {
		t.Run(c.name, func(t *testing.T) { boundaryPair(t, c) })
	}
}
func boundaryPair(t *testing.T, c boundaryCase) {
	s := boundaryServer(t, c)
	root := ensemble.New(log.New(io.Discard, "", 0))
	t.Cleanup(func() { _ = root.Shutdown() })
	a := boundaryAgent(t, root, ensemble.Gemini, boundaryModel, s.URL, false)
	var result ensemble.ResponseEvent
	// Both paths are checked against explicit canonical values, twice on ONE Engine.
	for repeat := 0; repeat < 2; repeat++ {
		for _, mode := range []string{"plain", "stream"} {
			result = boundaryCheck(t, a, s.URL+"/"+mode, c)
		}
	}
	boundaryReplay(t, root, a, result, c)
}
func TestChapter6GeminiBoundaryAnswerPrivacy(t *testing.T) {
	for _, c := range boundaryCases() {
		for _, disable := range []bool{false, true} {
			c := c
			t.Run(fmt.Sprintf("%s/plain=%t", c.name, disable), func(t *testing.T) {
				c.frames = [][]string{c.parts}
				s := boundaryServer(t, c)
				root := ensemble.New(log.New(io.Discard, "", 0))
				t.Cleanup(func() { _ = root.Shutdown() })
				// A catalogued requested model enables streaming; the response returns its own identity.
				a := boundaryAgent(t, root, ensemble.Gemini, "gemini-3.8-flash", s.URL, disable)
				ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
				defer cancel()
				answer, err := a.Ask(ctx, "answer with ordinary text only")
				if err != nil || answer != c.answer {
					t.Fatalf("Ask answer=%q err=%v want=%q", answer, err, c.answer)
				}
			})
		}
	}
}
