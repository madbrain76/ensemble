package main

import (
 "encoding/json"
 "fmt"
 "os"
 "path/filepath"
 "github.com/waywardgeek/ensemble/internal/grade"
 "github.com/waywardgeek/ensemble/internal/gradeedition2"
)
func main() {
 if len(os.Args)!=3 { panic("usage: full-api CORE CASE_JSON") }
 core, caseFile := os.Args[1],os.Args[2]
 example,err:=gradeedition2.ReadConsumerCase(caseFile); if err!=nil { panic(err) }
 bin,cleanup,err:=grade.Build(core); if err!=nil { panic(err) }; defer cleanup()
 fmt.Fprintf(os.Stderr,"built CLI %s from %s; Chapter10 starts\n",bin,core)
 media:=map[string]string{"anthropic":"claude-sonnet-4-6","openai":"gpt-5.4-mini","gemini":"gemini-3.8-flash"}
 report,err:=gradeedition2.Chapter10(bin,core,filepath.Join(core,"gui"),"claude-sonnet-4-6",filepath.Join(core,"ch05"),filepath.Join(core,"ch06"),filepath.Join(core,"ch07"),"claude-sonnet-4-6",example,media)
 if err!=nil { panic(err) }; if err=json.NewEncoder(os.Stdout).Encode(report);err!=nil { panic(err) }
}
